\section*{* §2.1\quad 连续性的证明与应用}

\noindent\textbf{注}\quad $\varepsilon-\delta$ 方法是数学院系学生的重点，非数学院系学生不作太高要求.

\subsection*{* 一、连续性的证明}

\noindent\textbf{要点}\quad 要证明一个函数 $f$ 在某区间 $I$ 上连续，只要在区间里任意取定一点 $x_0\in I$，证明 $\lim\limits_{x\to x_0}f(x)=f(x_0)$. 为此，我们可以

1）利用定义，证明：$\forall\varepsilon>0,\exists\delta>0$，当 $|x-x_0|<\delta$ 时，有 $|f(x)-f(x_0)|<\varepsilon$；

2）利用左、右极限，证明：$f(x_0+0)=f(x_0)=f(x_0-0)$；

3）利用序列语言，证明：$\forall\{x_n\}\to x_0$，有 $f(x_n)\to f(x_0)$；

4）利用邻域语言，证明：$\forall\varepsilon>0,\exists\delta>0$，使得
\[
  f((x_0-\delta,x_0+\delta))\subset (f(x_0)-\varepsilon,f(x_0)+\varepsilon);
\]

5）利用连续函数的运算性质：连续函数与连续函数经过有限次 $+,-,\cdot,\div$（除法要求除数不为零），复合（内层函数的值域在外层函数的定义域内），仍然是连续的.

\noindent\textbf{$\star$例 2.1.1}\quad 证明 Riemann 函数
\[
  R(x)=
  \begin{cases}
    \dfrac1q, & x=\dfrac pq\text{ 为既约分数},\ q>0,\\
    0, & x\text{ 为无理数}
  \end{cases}
\]
在无理点上连续，在有理点上间断.（浙江大学）

\noindent\textbf{证}\quad $1^\circ$ 设 $x_0$ 为有理点，$x_0=\dfrac pq$（为既约分数），$q>0$，则 $R(x_0)=\dfrac1q>0$. 由无理点的稠密性，存在无理点列 $\{x_n\}\to x_0$（当 $n\to\infty$ 时），但
\[
  |R(x_n)-R(x_0)|=\left|0-\frac1q\right|=\frac1q>0\quad(\forall n\in\mathbb{N}),
\]
即 $R(x_n)\not\to R(x_0)$. 故 $R(x)$ 在有理点不连续.

$2^\circ$（证明在无理点上连续.）设 $x_0\in[0,1]$ 为无理点，则 $R(x_0)=0$.

首先，我们从 $R(x)$ 的定义可以看出，$\forall\varepsilon>0,R(x)\ge\varepsilon$ 的点 $x$，在 $[0,1]$ 上最多只有有限个（事实上，要 $R(x)\ge\varepsilon>0$，$x$ 必须是有理点，若 $x=\dfrac pq,R\left(\dfrac pq\right)=\dfrac1q\ge\varepsilon$，则 $0\le p<q\le\dfrac1\varepsilon$. 可见满足此不等式的有理数 $\dfrac pq$ 最多只有有限个）. 如此，可取 $\delta>0$ 充分小，使得 $(x_0-\delta,x_0+\delta)$ 不含有 $R(x)\ge\varepsilon$ 之点，此即 $\forall x\in(x_0-\delta,x_0+\delta)$，有
\[
  |R(x)-R(x_0)|=R(x)<\varepsilon.
\]
这就证明了 $R(x)$ 在 $[0,1]$ 内的无理点上连续. 又因为 $R(x)$ 以 $1$ 为周期，\footnote{若 $x$ 为无理数，则 $R(x+1)=R(x)=0$；又若 $x=\dfrac pq$（$p,q$ 为互质整数），因 $1+x=\dfrac{q+p}{q}$，又 $(p+q)$ 与 $q$ 为互质整数，故 $x$ 为有理数时亦有 $R(x+1)=R(x)$. 总之，$R(x)$ 以 $1$ 为周期.}所以 $R(x)$ 在一切无理点上都连续.

\noindent\textbf{注}\quad $R(0)=1$，因为要使 $0=\dfrac pq$ 为既约分数，且 $q>0$，故只可能 $q=1,p=0$.

\noindent\textbf{$\star$例 2.1.2}\quad 设 $f(x)$ 在 $[a,b]$ 上连续，证明函数
\[
  M(x)=\sup_{a\le t\le x}f(t),\quad m(x)=\inf_{a\le t\le x}f(t)
\]
在 $[a,b]$ 上连续.（湖北大学）

（这里只就 $M(x)$ 进行证明，$m(x)$ 的连续性证明留作练习.）

\noindent\textbf{证}\quad 根据连续函数在闭区间上必达上、下确界的性质，$M(x)$ 在 $[a,b]$ 上处处有定义. 又因上确界随取值区间扩大而增大，知 $M(x)$ 严增. 故每点处的单侧极限存在. $\forall x_0\in[a,b]$，我们只要证明下面左、右等式分别在 $(a,b],[a,b)$ 成立即可：
\[
  M(x_0-0)=M(x_0)=M(x_0+0).
\tag{1}
\]
由 $M(x)$ 单调性，有 $M(x_0-0)\le M(x_0)$. 又因 $\forall x\in[a,x_0]$，有 $f(x)\le\sup\limits_{a\le t\le x}f(t)=M(x)\le M(x_0-0)$，所以 $M(x_0)=\sup\limits_{a\le t\le x_0}f(t)\le M(x_0-0)$，故 (1) 式左边等式成立.

下面用反证法证 (1) 中右边等式. 因 $M(x)$ 单调，$M(x_0)\le M(x_0+0)$. 假若 $M(x_0+0)>M(x_0)$，则可取充分小的 $\varepsilon_0>0$，使得 $M(x_0+0)>M(x_0)+\varepsilon_0$. 于是 $\forall x>x_0$，有
\[
  \sup_{a\le t\le x}f(t)=M(x)\ge M(x_0+0)>M(x_0)+\varepsilon_0.
\]
由确界定义，$\exists t\in[a,x]$，使得
\[
  f(t)>M(x_0)+\varepsilon_0\ge f(x_0)+\varepsilon_0,
\tag{2}
\]
但在 $[a,x_0]$ 上，$f(x)\le M(x_0)$，所以式 (2) 中的 $t\in(x_0,x]$. 这便与 $f(x)$ 的连续性矛盾. 证毕.

\noindent\textbf{例 2.1.3}\quad 设 $f(x)$ 在 $(0,1)$ 内有定义，且函数 $e^x f(x)$ 与 $e^{-f(x)}$ 在 $(0,1)$ 内都是单调不减的. 试证：$f(x)$ 在 $(0,1)$ 内连续.（北京师范大学）

\noindent\textbf{证}\quad $1^\circ$ 因 $e^{-f(x)}$ 单调不减，所以 $x>x_0$ 时，有 $e^{-f(x)}\ge e^{-f(x_0)}$，$e^{f(x_0)}\ge e^{f(x)}$，
\[
  f(x_0)\ge f(x).
\tag{1}
\]
此即表明 $f(x)$ 单调不增. 所以 $\forall x_0,f(x_0^+),f(x_0^-)$ 存在.

$2^\circ$ 由 $e^x f(x)$ 单调不减知：$x>x_0$ 时 $e^x f(x)\ge e^{x_0}f(x_0)$. 令 $x\to x_0^+$，得 $e^{x_0}f(x_0^+)\ge e^{x_0}f(x_0)$，
\[
  f(x_0^+)\ge f(x_0).
\tag{2}
\]

$3^\circ$ 在式 (1) 中，令 $x\to x_0^+$，得
\[
  f(x_0)\ge f(x_0^+).
\tag{3}
\]
式 (2) 和 (3) 表明 $f(x_0)=f(x_0^+)$. 类似可证 $f(x_0^-)=f(x_0)$. 从而 $f(x)$ 在 $x_0$ 处连续. 由 $x_0$ 的任意性，知 $f(x)$ 在 $(0,1)$ 内处处连续.

\noindent\textbf{*例 2.1.4}\quad 设 $f(x)$ 在 $(-\infty,+\infty)$ 内有定义，且

i）具有介值性（即：若 $f(x_1)<\mu<f(x_2)$，则 $\exists \xi\in(x_1,x_2)$，使得 $f(\xi)=\mu$）；

ii）对任意有理数 $r$，集合 $\{x\mid f(x)=r\}$ 为闭集.

试证：$f(x)$ 在 $(-\infty,+\infty)$ 上连续.

\noindent\textbf{证}\quad （反证法）若 $f$ 在某一点 $x_0$ 处不连续，则 $\exists\varepsilon_0>0$，使得 $\forall\dfrac1n>0,\exists x_n$，虽然
\[
  |x_n-x_0|<\frac1n,\quad \text{但}\quad |f(x_n)-f(x_0)|\ge\varepsilon_0.
\]
即 $\{x_n\}\to x_0$，但 $\{f(x_n)\}$ 在 $(f(x_0)-\varepsilon_0,f(x_0)+\varepsilon_0)$ 之外. 从而在 $(f(x_0)-\varepsilon_0,f(x_0)+\varepsilon_0)$ 之外至少一侧（例如右侧）含有 $\{f(x_n)\}$ 的无穷多项（如图 2.1.1 所示）：
\[
  f(x_{n_k})>f(x_0)+\varepsilon_0\quad(k=1,2,\cdots).
\]

在 $(f(x_0),f(x_0)+\varepsilon_0)$ 内任取一有理数 $r$：
\[
  f(x_0)<r<f(x_0)+\varepsilon_0<f(x_{n_k}).
\]
由介值性条件，对每一个 $x_{n_k}$，存在 $\xi_k$ 位于 $x_0$ 与 $x_{n_k}$ 之间，使得 $f(\xi_k)=r$（$k=1,2,\cdots$）. 因为 $x_{n_k}\to x_0$，所以 $\xi_k\to x_0$（当 $k\to\infty$ 时）. 这表明 $x_0$ 是 $\{x\mid f(x)=r\}$ 的一个聚点. 由已知条件 (2)，知 $x_0\in\{x\mid f(x)=r\}$，即 $f(x_0)=r$，与 $f(x_0)<r$ 矛盾，证毕.

\noindent\textbf{例 2.1.5}\quad 设函数 $y=f(x)$ 在 $(a,b)$ 内有定义，具有介值性，并且是一对一的（即若 $x_1\ne x_2$，则必有 $f(x_1)\ne f(x_2)$）. 试证：

1）$f(x)$ 是严格单调的，值域为某个开区间 $J$；

2）$f^{-1}(y)$ 在 $J$ 内单调，而且也有介值性；

3）$f(x),f^{-1}(y)$ 连续.

\noindent\textbf{证}\quad 1）由 $f$ 是一对一的，假若 $f$ 不严格单调，则必 $\exists x_1<x_2<x_3$ 使得
\[
  f(x_1)<f(x_2),\quad f(x_2)>f(x_3)
\]
或
\[
  f(x_1)>f(x_2),\quad f(x_2)<f(x_3).
\]
下面只就前一种情况进行讨论，后一种情况类似可证.

任取一数 $\mu$，使得 $\max\{f(x_1),f(x_3)\}<\mu<f(x_2)$. 由介值性知：$\exists\xi_1\in(x_1,x_2)$，$\xi_2\in(x_2,x_3)$，使得
\[
  f(\xi_1)=\mu=f(\xi_2),
\]
如图 2.1.2，这就和 $f$ 是一对一的条件矛盾，故 $f$ 只能严格单调. 为了确定起见，下面不妨假设 $f$ 严增. 由介值性，显然 $f$ 的值必填满某区间，（可以为无穷区间，但必为开区间！）记为 $J$.

\noindent\textbf{*例 2.1.6}\quad 证明定理：$f(x)$ 在实轴 $X$ 上连续 $\Leftrightarrow$ 任何开集的逆像仍为开集（即：设 $O$ 为 $Y$ 轴上的开集，则 $f^{-1}(O)=\{x\mid f(x)\in O\}$ 为 $X$ 轴上的开集）.\footnote{集合 $O$ 被称为开集，指它的每个点皆为内点，即 $\forall x_0\in O,\exists\delta>0$，使得 $(x_0-\delta,x_0+\delta)\subset O$.}

\noindent\textbf{证}\quad $1^\circ$（$\Rightarrow$）要证 $f^{-1}(O)$ 为 $X$ 轴的开集，即要证明：$\forall x_0\in f^{-1}(O),\exists\delta>0$ 使
\[
  (x_0-\delta,x_0+\delta)\subset f^{-1}(O).
\tag{1}
\]
由 $x_0\in f^{-1}(O)$，知 $y_0=f(x_0)\in O$. 既然 $O$ 为开集，所以 $\exists\varepsilon>0$，使得
\[
  (f(x_0)-\varepsilon,f(x_0)+\varepsilon)=(y_0-\varepsilon,y_0+\varepsilon)\subset O.
\tag{2}
\]
由于 $f$ 连续，对 $y_0$ 的邻域 $(y_0-\varepsilon,y_0+\varepsilon)$，$\exists\delta>0$，使得
\[
  f((x_0-\delta,x_0+\delta))\subset (y_0-\varepsilon,y_0+\varepsilon)\subset O,
\]
从而 $(x_0-\delta,x_0+\delta)\subset f^{-1}((y_0-\varepsilon,y_0+\varepsilon))\subset f^{-1}(O)$，故 $f^{-1}(O)$ 为开集.

$2^\circ$（$\Leftarrow$）已知任何开集的逆像仍为开集，故 $\forall x_0\in X,\forall\varepsilon>0$，（设 $y_0=f(x_0)$）$(y_0-\varepsilon,y_0+\varepsilon)$ 的逆像 $f^{-1}((y_0-\varepsilon,y_0+\varepsilon))$ 为开集. 由此对于 $x_0\in f^{-1}((y_0-\varepsilon,y_0+\varepsilon))$，$\exists\delta>0$，使得 $(x_0-\delta,x_0+\delta)\subset f^{-1}((y_0-\varepsilon,y_0+\varepsilon))$. 故
\[
  f((x_0-\delta,x_0+\delta))\subset (y_0-\varepsilon,y_0+\varepsilon)=(f(x_0)-\varepsilon,f(x_0)+\varepsilon),
\]
所以 $f$ 连续.

\noindent\textbf{评述}\quad 该定理具有重大意义，因为它实际上给出了连续性的另一种新的定义方式，这种方式可以摆脱 $\varepsilon-\delta$，利用邻域、开集的工具，建立抽象空间里的连续映射理论. 这正是后来点集拓扑学的思想渊源.

\noindent\textbf{*例 2.1.7}\quad 设 $f(x)$ 在 $(-\infty,+\infty)$ 上有定义，证明：$f(x)$ 连续 $\Leftrightarrow \forall c\in(-\infty,+\infty)$，集合 $\{x\mid f(x)>c\}$ 与 $\{x\mid f(x)<c\}$ 为开集.

\noindent\textbf{提示}\quad 必要性可利用连续函数保号性证明.（充分性）$\forall x_0\in(-\infty,+\infty),\forall\varepsilon>0,x_0\in\{x\mid f(x)>f(x_0)-\varepsilon\}$ 为开集.

\noindent\textbf{$\star$例 2.1.8}\quad 1）证明：若函数 $f(x),g(x)$ 连续，则函数 $\varphi(x)=\min\{f(x),g(x)\}$，$\psi(x)=\max\{f(x),g(x)\}$ 亦连续；

2）设 $f_1(x),f_2(x),f_3(x)$ 在 $[a,b]$ 上连续，令函数 $f$ 的值 $f(x)$ 等于三值 $f_1(x),f_2(x),f_3(x)$ 中介于其他两值之间的那个值，证明 $f$ 在 $[a,b]$ 上连续；（西安电子科技大学）

3）令
\[
  u_n(x)=
  \begin{cases}
    -n, & \text{当 }x\le -n\text{ 时},\\
    x, & \text{当 }-n<x\le n\text{ 时},\\
    n, & \text{当 }x>n\text{ 时},
  \end{cases}
\]
$f(x)$ 为实函数，试证明：$f(x)$ 连续的充要条件是 $g_n(x)=u_n[f(x)]$ 对任意固定的 $n$，都是 $x$ 的连续函数.（四川大学）

\noindent\textbf{证}\quad 1）
\[
  \varphi(x)=\frac{f(x)+g(x)-|f(x)-g(x)|}{2},
\]
\[
  \psi(x)=\frac{f(x)+g(x)+|f(x)-g(x)|}{2}.
\]

2）$f(x)=f_1(x)+f_2(x)+f_3(x)-\max\{f_1(x),f_2(x),f_3(x)\}-\min\{f_1(x),f_2(x),f_3(x)\}$.

3）
\[
\begin{aligned}
g_n(x)=u_n[f(x)]
&=-n+f(x)+n-\max\{-n,f(x),n\}\\
&\quad-\min\{-n,f(x),n\}.
\end{aligned}
\]
\[
\begin{aligned}
&=f(x)-\max\{f(x),n\}-\min\{f(x),-n\}\\
&=f(x)-\frac{n+f(x)+|f(x)-n|}{2}
  -\frac{-n+f(x)-|n+f(x)|}{2}\\
&=\frac{|n+f(x)|-|n-f(x)|}{2}.
\end{aligned}
\]
以上由连续函数的运算性质，即知它们连续. 3）的充分性留作练习.

  f\left(\frac{x+y}{2}\right)\le \frac{f(x)+f(y)}{2},
\tag{1}

\subsection*{二、连续性的应用}

上段我们主要讨论如何由给定的条件，证明函数连续. 现在我们要讨论相反的问题：假定所讨论的函数连续，证明在某些条件下，有什么结果；或者构造适当的函数，把别的问题转化为连续函数的问题.

\noindent\textbf{*例 2.1.10}\quad 证明：（非常数的）连续周期函数必有最小正周期.（南开大学，南京大学）

\noindent\textbf{分析}\quad 若有最小正周期 $T_0$，那么 $T_0$ 便是所有正周期的下确界. 反之，若能证明全体正周期的下确界仍为一个正周期，则这个正周期自然是最小正周期. 因此我们的问题只要证明如下三点即可：$1^\circ\inf\{f\text{ 的正周期}\}=T_0$ 存在；$2^\circ T_0$ 仍为 $f$ 的周期；$3^\circ T_0>0$.

\noindent\textbf{证}\quad $1^\circ$ 因为集合 $\{f\text{ 的正周期}\}$ 有下界 $0$，根据确界存在定理，$\inf\{f\text{ 的正周期}\}=T_0$ 存在.

$2^\circ$ 证明 $T_0\in\{f\text{ 的周期}\}$. 根据确界性质，$\exists T_n\in\{f\text{ 的正周期}\}$（$n=1,2,\cdots$），使 $T_n\to T_0$（$n\to\infty$）. 如此，$\forall x\in\mathbb{R}$，有
\[
  f(x+T_0)=f\left(x+\lim_{n\to\infty}T_n\right)
  =\lim_{n\to\infty}f(x+T_n)=f(x),
\]
此式表明 $T_0$ 是 $f$ 的周期.

$3^\circ$ 因 $T_n>0,T_n\to T_0$（$n\to\infty$），所以 $T_0\ge0$. 假若 $T_0=0$，则 $T_n\to0$（$n\to\infty$），$f$ 的周期网点（指等于周期整数倍的点）在实轴 $\mathbb{R}$ 上稠密. 从而，$\forall x\in\mathbb{R},\exists\{x_n\}\to x$（其中 $\{x_n\}$ 是由一些周期网点所组成的序列）. 于是
\[
  f(x)=f\left(\lim_{n\to\infty}x_n\right)=\lim_{n\to\infty}f(x_n)
  =\lim_{n\to\infty}f(0+x_n)=f(0),
\]
即 $f(x)\equiv f(0)$（常数），矛盾. 故 $T_0>0$.

\noindent\textbf{注}\quad 关于周期的几点补充：

$1^\circ$ 若 $T$ 和 $T_1$ 是函数 $f$ 的周期，则 $mT+nT_1$ 仍是 $f$ 的周期.

$2^\circ$ 若 $f(\ne\text{ 常数})$ 是连续函数，有最小正周期 $T_0$（例 2.1.10），那么，$\forall T\in\{f\text{ 的正周期}\}$，必 $\exists n\in\mathbb{N}$ 使得 $T=nT_0$.

（因为：若该论断不成立，则存在正周期 $T>T_0,\forall n\in\mathbb{N},T\ne nT_0,\exists n\in\mathbb{N}$，使得 $nT_0<T<(n+1)T_0$，因而，$T_0>T-nT_0>0$. 至此，我们找到比 $T_0$ 还小的正周期：$T-nT_0$，跟“$T_0$ 为最小正周期”矛盾.）

$3^\circ$ 设 $f(\ne\text{ 常数})$ 是连续函数，有周期 $T>0$. 若 $\alpha T$ 也是 $f$ 的周期，则 $\alpha$ 最多是有理数.（不可能是无理数！）

（因为：若 $\alpha T$ 是 $f$ 的周期，则（如 $2^\circ$ 所述）有 $m\in\mathbb{N}$，使得 $\alpha T=mT_0$；有 $n\in\mathbb{N}$ 使得 $T=nT_0$. 可见 $\alpha nT_0=mT_0,\alpha=\dfrac mn$ 为有理数.）

\noindent\textbf{$\star$例 2.1.11}\quad 设 $f(x)$ 对 $(-\infty,+\infty)$ 内一切 $x$ 有
\[
  f(x^2)=f(x),
\tag{1}
\]
且 $f(x)$ 在 $x=0,x=1$ 处连续，证明 $f(x)$ 在 $(-\infty,+\infty)$ 为常数.（华东师范大学）

\noindent\textbf{证}\quad $1^\circ$ 设 $x>0$，由式 (1)，
\[
  f(x)=f(x^{1/2})=f(x^{1/4})=\cdots=f(x^{1/2^n})=\cdots,
\]
因此
\[
  f(x)=\lim_{n\to\infty}f(x^{1/2^n})
  =f\left(\lim_{n\to\infty}x^{1/2^n}\right)=f(1).
\]

$2^\circ$ 当 $x<0$ 时，$f(x)=f(x^2)=f(1)$.

$3^\circ$ 当 $x=0$ 时，$f(0)=\lim\limits_{x\to0}f(x)=f(1)$.

故 $f(x)\equiv f(1)$（常数）.

\noindent\textbf{例 2.1.12}\quad 设 $f:[0,1]\to[0,1]$ 为连续函数，$f(0)=0,f(1)=1,f(f(x))=x$，试证 $f(x)=x$.

\noindent\textbf{分析}\quad $1^\circ$ 要证 $f(x)=x$，只需证 $f(x)$ 严增. 实际上，若证明了 $f(x)$ 严增，利用 $f(f(x))=x$，立即可证 $f(x)=x$. 因 $\forall x\in[0,1]$，要么 $f(x)\ge x$，要么 $f(x)\le x$. 由 $f(x)$ 严增知 $f(x)\ge x$ 时，有 $x=f(f(x))\ge f(x)$；$f(x)\le x$ 时，有 $x=f(f(x))\le f(x)$. 故总有 $f(x)=x$. 问题归纳为证明 $f(x)$ 严增.

$2^\circ$ 例 2.1.5 告诉我们，有介值性与一对一性就可得到单调性；再利用 $f(0)=0,f(1)=1$，便可得 $f(x)$ 严增. 剩下问题在于证明 $f$ 为一对一的. 事实上，$\forall x_1,x_2\in[0,1]$，若 $f(x_1)=f(x_2)$，则利用条件 $f(f(x))=x$ 可知
\[
  x_1=f(f(x_1))=f(f(x_2))=x_2,
\]
故 $f$ 为一对一的.

\noindent\textbf{new $\star$练习}\quad 设 $f$ 是从 $\mathbb{R}$ 到 $\mathbb{R}$ 的一对一连续映射，有不动点，满足：
\[
  f(2x-f(x))=x,\quad \forall x\in\mathbb{R}.
\]
试证：$f(x)\equiv x$（$\forall x\in\mathbb{R}$）.

\noindent\textbf{证}\quad 明显：$f(x)\equiv x$（$\forall x\in\mathbb{R}$）是问题的一个解. 设 $g(x)$ 是问题的另一个解，那么 $\forall x\in\mathbb{R}$，
\[
  g(2x-g(x))=x=f(2x-f(x)).
\]
因为 $f(x)$ 和 $g(x)$ 都是一对一的连续函数，所以 $2x-g(x)=2x-f(x)$，即
\[
  g(x)=f(x)=x\quad(\forall x\in\mathbb{R}),
\]
故 $f(x)\equiv x$ 是唯一解. 因此 $f$ 为一对一的.

\noindent\textbf{new *例 2.1.18}\quad 设 $f:[0,1]\to\mathbb{R}$ 单调递增，且 $f([0,1])$ 是闭集，证明：$f$ 在 $[0,1]$ 上连续.（中国科学技术大学）

\noindent\textbf{提示}\quad 证明 $f$ 在 $(0,1]$ 上处处左连续（类似可证 $[0,1)$ 上处处右连续）.

\noindent\textbf{证}\quad $1^\circ$ $\forall x_0\in(0,1],\forall\{x_n\}\subset[0,1],x_n\nearrow x_0$，下面要证明：$\lim\limits_{n\to\infty}f(x_n)=f(x_0)$.

因为 $f$ 递增，知 $\{f(x_n)\}$ 递增，且有上界 $f(x_0)$，由单调有界原理，$\lim\limits_{n\to\infty}f(x_n)$ 存在，记作
\[
  \lim_{n\to\infty}f(x_n)=A,
\tag{1}
\]
且
\[
  f(x_n)\le A\le f(x_0).
\tag{2}
\]

$2^\circ$ 已知 $f([0,1])$ 是闭集（意指：含它的所有极限点）. 式 (1) 表明：$A$ 是 $\{f(x_n)\}\subset f([0,1])$ 的极限点，由 $f([0,1])$ 是闭集知 $A\in f([0,1])$，即 $\exists a\in[0,1]$ 使得
\[
  f(a)=A.
\tag{3}
\]
由式 (2) 知 $x_n\le a\le x_0$（$\forall n\in\mathbb{N}$），即 $0\le x_0-a\le x_0-x_n\to0$（$n\to\infty$），可得
\[
  a=x_0.
\tag{4}
\]
因此
\[
  \lim_{n\to\infty}f(x_n)\overset{(1)}=A\overset{(3)}=f(a)\overset{(4)}=f(x_0).
\tag{5}
\]
根据 $x_0\in(0,1]$ 和 $\{x_n\}\subset[0,1]$ 的任意性，式 (5) 表明：$f(x)$ 在 $(0,1]$ 上处处左连续（类似可证 $[0,1)$ 上处处右连续）.

\noindent\textbf{new 例 2.1.19}\quad 设 $y=f(x)$ 是从区间 $[0,1]$ 映射到 $[0,1]$ 的函数，其图像
\[
  E=\{(x,y)\mid y=f(x),x\in[0,1]\}
\]
是单位正方形 $[0,1]\times[0,1]$ 上的闭集，求证：$f$ 是连续函数.（中国科学技术大学）

（尚无“二元函数”知识者，此题暂缓.）

\noindent\textbf{提示}\quad 跟上题比较，无单调条件；但多了有界条件，因而可用致密性定理.

\noindent\textbf{证}\quad （反证法）假设 $f$ 在某点 $x_0\in[0,1]$ 不连续，意味着 $\exists\varepsilon_0>0,\forall n\in\mathbb{N},\delta_n=\dfrac1n>0,\exists x_n\in[0,1]$，虽然 $|x_n-x_0|<\delta_n=\dfrac1n$（即 $\exists\{x_n\}\to x_0$），但
\[
  |f(x_n)-f(x_0)|\ge\varepsilon_0\quad(n=1,2,\cdots).
\tag{1}
\]
另一方面，根据致密性定理，有界无穷序列 $\{f(x_n)\}_{n=1}^{\infty}\subset[0,1]$ 必有收敛子列，即存在 $\{f(x_{n_k})\}\subset\{f(x_n)\}$，使得 $\lim\limits_{k\to\infty}f(x_{n_k})$ 存在，记作 $y_0$. 而 $\lim\limits_{k\to\infty}x_{n_k}=x_0$，说明 $(x_0,y_0)$ 是 $\{(x_{n_k},f(x_{n_k}))\}$ 的极限点，且 $\{(x_{n_k},f(x_{n_k}))\}\subset E$，亦即 $(x_0,y_0)$ 是 $E$ 的一个极限点，但 $E$ 是闭集（包含它的所有极限点）. 故
\[
  (x_0,y_0)\in E=\{(x,y)\mid y=f(x),x\in[0,1]\}\subset[0,1]\times[0,1].
\]
这表明 $y_0=f(x_0)$. 于是
\[
  \lim_{k\to\infty}f(x_{n_k})=y_0=f(x_0).
\tag{2}
\]
注意 $\{(x_{n_k},f(x_{n_k}))\}$ 是 $\{(x_n,f(x_n))\}$ 的子列，应满足式 (1)，故 $\lim\limits_{k\to\infty}f(x_{n_k})\ne f(x_0)$，与式 (2) 矛盾. 连续性获证.

\noindent\textbf{new $\star$例 2.1.21}\quad 是否存在 $[a,b]$ 上的连续函数：在有理点上取值为无理数，在无理点上为有理数？请作出判断并说明理由.（南开大学）

\noindent\textbf{“势”的简介：}\quad 一区间里的有理数和无理数都是无穷多个，无法再比较. 作为“数目个数”的推广，数学家引入了“势”的概念；两个集合如果能建立一一对应关系，则称它们具有对等的“势”. 能跟自然数一一对应的集合，称为可数集；否则称为不可数集. 结果发现：全体有理数是可数集，全体无理数是不可数集.

任意有限个可数集之并仍是可数集，可数个可数集的并还是可数集.

全体实数的“势”称为连续统的势，与任何（非退化的）区间 $I$（不论有限无限，闭或开）的“势”是对等的，也与 $I$ 中无理数集的“势”对等. 跟连续统的势相比，有理数的势似乎小得可以忽略不计.

\noindent\textbf{解答}\quad （根据上面的理念）题中的连续函数不可能存在！原因是

$1^\circ$ 在有限闭区间 $[a,b]$ 上的连续函数，必达最大、最小值，并有介值性. 因此 $f$ 的像点填满某个闭区间，即 $\exists[A,B]$ 使得 $f([a,b])=[A,B]$.

$2^\circ$ 函数 $f$ 可以是多对一或一对一.（非一对多！）因此，像点的势不会超过像源的势. 按题设：
\[
  f(\text{有理数})=\text{无理数}\quad(\text{因为像源是可数集，故像点最多是可数集}),
\]
\[
  f(\text{无理数})=\text{有理数}\quad(\text{有理数是可数集}).
\]
可数集加可数集还是可数集. 故 $f$ 的全体像点 $f([a,b])$ 是可数集，不可能填满
任一个区间，即 $\forall[A,B],f([a,b])\ne[A,B]$. 与 $1^\circ$ 矛盾.

\noindent\textbf{new 例 2.1.22}\quad 设函数 $f(x)$ 在 $(-\infty,+\infty)$ 上连续，$n$ 为奇数. 若

  \lim_{x\to+\infty}\frac{f(x)}{x^n}
  =
  \lim_{x\to-\infty}\frac{f(x)}{x^n}=1,

\noindent\textbf{new *例 2.1.24（定理）}\quad $f$ 是 $[a,b]\to[a,b]$ 上的连续函数，设 $x_1\in[a,b],x_{n+1}=f(x_n)$，$\forall n\in\mathbb{N}$，证明：$\{x_n\}$ 收敛的充分必要条件是
\[
  \lim_{n\to\infty}(x_{n+1}-x_n)=0.
\tag{1}
\]
（源自《美国数学月刊》）

\noindent\textbf{证}\quad 根据 Cauchy 准则，式 (1) 的必要性明显. 只需证明充分性. 充分性的证明其实也不难，需明确三点：

$1^\circ$ 所谓某点是序列 $\{x_n\}$ 的极限点，意指：它是 $\{x_n\}$ 的某子列的极限.

$2^\circ$ 根据致密性原理：“有界无穷序列 $\{x_n\}$ 必有极限点”.

$3^\circ$ 数列 $\{x_n\}:\lim\limits_{n\to\infty}x_n=\alpha\Leftrightarrow\alpha$ 是 $\{x_n\}$ 的唯一极限点.

（事实上，若 $\{x_n\}$ 收敛于 $\alpha$，则 $\{x_n\}$ 的任何子列都以 $\alpha$ 为极限. 因此，收敛数列只有唯一极限点. 反之，若 $\{x_n\}$ 只有唯一极限点 $\alpha$，则 $\{x_n\}$ 必以 $\alpha$ 为极限. 这因为：倘若 $\{x_n\}$ 不以 $\alpha$ 为极限，则 $\alpha$ 有某邻域，在其外必有 $\{x_n\}$ 的无穷多项 $\{x_{n_k}\}$，根据致密性原理，$\{x_{n_k}\}$ 至少另有一个极限点 $\beta\ne\alpha$，与极限点唯一矛盾.）

（下面（用反证法）证明式 (1) 的充分性.）

设 $\{x_n\}$ 不收敛，如 $3^\circ$ 所述，$\{x_n\}$ 至少有两个不同的极限点，记作 $\alpha_1$ 和 $\alpha_2$. 对 $\alpha_1$ 和 $\alpha_2$，各作一微小邻域 $U(\alpha_1,\varepsilon)$ 和 $U(\alpha_2,\varepsilon)$，使得两邻域无公共点. 既然 $\alpha_1$ 和 $\alpha_2$ 都是 $\{x_n\}$ 的极限点，因此对任意 $N>0$（不论多么大），总 $\exists n_k>N$，使得 $x_{n_k}\in U(\alpha_1,\varepsilon)$；同理，又能找到更大的 $n_{k+1}>n_k$，使得 $x_{n_{k+1}}\in U(\alpha_2,\varepsilon)$，继而找到 $n_{k+2}>n_{k+1}$，使得 $x_{n_{k+2}}\in U(\alpha_1,\varepsilon),\cdots$. 总之，$\{x_n\}$ 会不停地在两邻域来回摆动.

（i）（下面证明：$\alpha_1$ 和 $\alpha_2$ 之间的每点 $\eta$ 都必是 $\{x_n\}$ 的极限点.）设 $\eta\in(\alpha_1,\alpha_2)$ 是 $\alpha_1$ 和 $\alpha_2$ 之间的任意一点，$\forall\varepsilon>0$，作邻域 $U(\eta,\varepsilon)$. 如上所述：$\{x_n\}$ 在 $\alpha_1$ 和 $\alpha_2$ 之间来回摆动，每趟 $\{x_n\}$ 都要经过邻域 $U(\eta,\varepsilon)$. 但 $\lim\limits_{n\to\infty}(x_{n+1}-x_n)=0$，说明 $\{x_n\}$ 前进的步幅无限变小. 当步幅小于 $\varepsilon$ 时，经过邻域 $U(\eta,\varepsilon)$ 时至少有一点落入 $U(\eta,\varepsilon)$ 中. 无穷次摆动，$U(\eta,\varepsilon)$ 里将含有 $\{x_n\}$ 的无穷多项，由 $\varepsilon>0$ 的任意性，说明 $\eta$ 是 $\{x_n\}$ 的极限点. 再由 $\eta(\eta\in(\alpha_1,\alpha_2))$ 的任意性，说明：$\alpha_1$ 和 $\alpha_2$ 之间的每一点都是 $\{x_n\}$ 的极限点.

（ii）（证明：$\alpha_1$ 和 $\alpha_2$ 之间的每一点 $\eta$ 都是 $f$ 的不动点.）上面已证 $\eta$ 是极限点，因此存在子列 $\{x_{n_k}\}\to\eta$（当 $k\to\infty$ 时）. 由于 $\lim\limits_{n\to\infty}(x_{n+1}-x_n)=0$，
\[
  |x_{n_k+1}-\eta|\le |x_{n_k+1}-x_{n_k}|+|x_{n_k}-\eta|\to0,
\]
知
\[
  \{x_{n_k+1}\}\to\eta.
\]
于是，在等式 $f(x_{n_k})=x_{n_k+1}$ 里令 $k\to\infty$，取极限得 $f(\eta)=\eta$.

这就证明了：$\alpha_1$ 和 $\alpha_2$ 之间的每一点 $\eta$ 都是 $f$ 的不动点.

既然区间 $[\alpha_1,\alpha_2]$ 上每点都是不动点，那么 $\{x_n\}$ 第一次落入区间 $(\alpha_1,\alpha_2)$，就不能再动，与无穷次来回摆动相矛盾. 故 $\{x_n\}$ 收敛. 证毕.

\noindent\textbf{注}\quad 该命题虽然将收敛条件作了大幅度的弱化，但适应面较窄，只针对有限闭区间里的递推数列，且递推公式连续才成立. 因为该定理尚未列入教材，考生千万慎用.

\noindent\textbf{再提示}\quad （$\Leftarrow$）$f$ 在 $[0,1]$ 上连续 $\Rightarrow \exists x_0\in[0,1]$，使 $f(x_0)=x\in\max\limits_{[0,1]}f(x)=x\in\max\limits_{[0,x_0]}f(x)=x\in\sup\limits_{[0,x_0]}f(x)=R(x_0)\Rightarrow G(x_0)=1$. 若 $f(x)<R(x)$，则 $G(x)=0$，故 $G$ 只可能取 $0,1$ 两值. 既然 $G(x_0)=1$，且 $G(x)$ 在 $[0,1]$ 上连续，故 $G(x)\equiv1$. 从而 $f(x)=R(x)$，知 $f(x)\nearrow$.

\noindent\textbf{*2.1.4}\quad 设函数 $f(x)$ 在 $[a,b]$ 上连续且恒大于零，按 $\varepsilon-\delta$ 定义证明：$\dfrac1{f(x)}$ 在 $[a,b]$ 上连续.（中南大学）

\noindent\textbf{注}\quad 该题虽是基本题，但不易叙述严整.

\noindent\textbf{证 I}\quad 因 $f>0$ 于 $[a,b]$ 上连续，知 $\exists x^*\in[a,b]$，使 $0<f(x^*)=\min\limits_{[a,b]}f\overset{\text{记}}=m$，故 $\forall x_0\in[a,b]$，$\forall\varepsilon>0$，取 $\delta>0$，使得 $|x-x_0|<\delta,x\in[a,b]$ 时，有 $|f(x)-f(x_0)|<m^2\varepsilon$，
\[
  \left|\frac1{f(x)}-\frac1{f(x_0)}\right|
  =\left|\frac{f(x)-f(x_0)}{f(x)f(x_0)}\right|
  \le\frac1{m^2}|f(x)-f(x_0)|<\varepsilon.
\]
所以 $\dfrac1f$ 在 $[a,b]$ 上连续.

\noindent\textbf{证 II}\quad $\forall x_0\in[a,b]$，$\exists\delta_1>0$，使得 $x\in U(x_0,\delta_1)\cap[a,b]$ 时，
\[
  |f(x)-f(x_0)|<\frac{|f(x_0)|}{2},
\]
从而
\[
  |f(x)|\ge\bigl||f(x_0)|-|f(x)-f(x_0)|\bigr|\ge \frac{|f(x_0)|}{2}.
\]
$\forall\varepsilon>0,\exists\delta_2>0$，使得 $x\in U(x_0,\delta_2)\cap[a,b]$ 时，
\[
  |f(x)-f(x_0)|<\frac{(f(x_0))^2}{2}\cdot\varepsilon.
\]
于是取 $\delta=\min\{\delta_1,\delta_2\}$，则当 $x\in U(x_0,\delta)\cap[a,b]$ 时有
\[
  \left|\frac1{f(x)}-\frac1{f(x_0)}\right|
  \le\frac{|f(x)-f(x_0)|}{|f(x)\cdot f(x_0)|}
  \le\frac2{(f(x_0))^2}|f(x)-f(x_0)|<\varepsilon.
\]
故 $f^{-1}$ 在 $[a,b]$ 上连续.

\noindent\textbf{注}\quad 证 I 只适用于有界闭区间；证 II 适用于一切区间，且不必要求 $f>0$. 只要 $f(x_0)\ne0,f$ 就在 $x_0$ 处连续.

\noindent\textbf{*2.1.17}\quad 若函数 $f(x)$ 在 $D$ 上有界，令
\[
  M_f(x_0,\delta)=\sup\{|f(x)|\mid x\in D,\ |x-x_0|<\delta\},
\]
\[
  m_f(x_0,\delta)=\inf\{|f(x)|\mid x\in D,\ |x-x_0|<\delta\},
\]
证明：

1）当 $\delta\to0^+$ 时，$M_f(x_0,\delta)-m_f(x_0,\delta)$ 的极限存在；

2）函数 $f(x)$ 在 $x_0$ 处连续的充要条件是
\[
  \lim_{\delta\to0^+}[M_f(x_0,\delta)-m_f(x_0,\delta)]=0.
\]
（西北大学）

\noindent\textbf{提示}\quad 1）（注意：随着取值范围减小，$\sup(\cdot)\searrow,\inf(\cdot)\nearrow$）$\delta\searrow0$ 时，$M_f(x_0,\delta)-m_f(x_0,\delta)\searrow$，且有下界 $0$.

2）（$\Leftarrow$）$|f(x)-f(x_0)|\le M_f-m_f\to0$.

（$\Rightarrow$）$f$ 在 $x_0$ 处连续 $\Rightarrow \forall\varepsilon>0,\exists\delta>0$，当 $|x-x_0|<\delta,x\in D$ 时，
\[
  f(x_0)-\frac\varepsilon4<f(x)<f(x_0)+\frac\varepsilon4.
\]
取 $\sup(\cdot)$ 得
\[
  f(x_0)-\frac\varepsilon2<f(x_0)-\frac\varepsilon4<f(x)\le M_f(x_0,\delta)
  \le f(x_0)+\frac\varepsilon4<f(x_0)+\frac\varepsilon2.
\]
即
\[
  |M_f(x_0,\delta)-f(x_0)|<\frac\varepsilon2.
\]
类似有
\[
  |f(x_0)-m_f(x_0,\delta)|<\frac\varepsilon2.
\]
于是
\[
  |M_f(x_0,\delta)-m_f(x_0,\delta)|
  \le |M_f-f(x_0)|+|f(x_0)-m_f|<\frac\varepsilon2+\frac\varepsilon2=\varepsilon.
\]

  \end{cases}
\]
$y=2,3,\cdots$ 时右连续，左不连续；$[0,+\infty)$ 上其余处处连续.

\noindent\textbf{*2.1.20}\quad 设 $f(x)$ 在 $[0,1]$ 上定义，并且有界，$a,b>1$ 为两常数，当 $0\le x\le\dfrac1a$ 时，有 $f(ax)=bf(x)$，试证 $f$ 在 $x=0$ 处右连续.

\noindent\textbf{提示}\quad 在 $f(ax)=bf(x)$ 中令 $x=0$，得 $f(0)=0$. 因此只要证明 $\lim\limits_{x\to0^+}f(x)=0$ 即可.

\noindent\textbf{再提示}\quad 因 $0\le x\le\dfrac1a$ 时，有 $f(ax)=bf(x)$，即 $f(x)=\dfrac{f(ax)}b$，因此 $\forall n\in\mathbb{N}$（自然数），$0\le x\le\dfrac1{a^n}$ 时，有
\[
  f(x)=\frac{f(ax)}b=\cdots=\frac{f(a^n x)}{b^n},
\]
\[
  |f(x)|=\frac{|f(a^n x)|}{b^n}\le\frac{M}{b^n}
\]
（因 $f$ 有界，即 $\exists M>0$，使 $|f(x)|\le M$）. 故 $\forall\varepsilon>0$，取定 $N$（充分大）可使 $\dfrac{M}{b^N}<\varepsilon$，然后令 $\delta=\dfrac1{a^N}$，则当 $0\le x\le\delta=\dfrac1{a^N}$ 时，有 $|f(x)|\le\dfrac{M}{b^N}<\varepsilon$. 即 $\lim\limits_{x\to0^+}f(x)=0=f(0)$.

\noindent\textbf{**2.1.21}\quad 设 $y=f(x)$ 为 $X\to Y$ 的连续函数，$F$ 为 $Y$ 轴上的闭集，试证 $f^{-1}(F)$ 为 $X$ 轴上的闭集.

\noindent\textbf{证}\quad 要证 $f^{-1}(F)$ 为 $X$ 轴上的闭集，即要证 $f^{-1}(F)$ 包含它的一切聚点. 设 $x_0$ 是 $f^{-1}(F)$ 的任一聚点（指：$\exists x_n\in f^{-1}(F),x_n\to x_0(n\to\infty)$），来证 $x_0\in f^{-1}(F)$. 事实上：$x_n\in f^{-1}(F)\Rightarrow f(x_n)\in F \overset{\text{连续性}}{\Longrightarrow}\lim\limits_{n\to\infty}f(x_n)=f(x_0)\Rightarrow f(x_0)$ 是 $F$ 的聚点 $\overset{\text{因为 }F\text{ 为闭集}}{\Longrightarrow}f(x_0)\in F\Rightarrow x_0\in f^{-1}(F)$. 由 $x_0$ 的任意性，知 $f^{-1}(F)$ 为 $X$ 上的闭集.

\noindent\textbf{*2.1.22}\quad 函数 $f,g$ 在 $[a,b]$ 上连续，$f$ 单调，$x_n\in[a,b]$ 使得 $g(x_n)=f(x_{n+1})$（$n=1,2,\cdots$），证明：$\exists x_0\in[a,b]$，使得 $f(x_0)=g(x_0)$.

\noindent\textbf{证}\quad $1^\circ$ 若 $\exists n$ 使得 $g(x_n)-f(x_n)$ 与 $g(x_{n+1})-f(x_{n+1})$ 异号，则由连续函数介值性，知 $\exists x_0\in[a,b]$，使得 $g(x_0)=f(x_0)$.

$2^\circ$ 否则 $|g(x_n)-f(x_n)|$ 保持同号. 例如恒正，则 $f(x_{n+1})-f(x_n)=g(x_n)-f(x_n)>0$（$n=1,2,\cdots$），表明 $\{f(x_n)\}\nearrow$. 由 $f$ 的单调性，知 $\{x_n\}$ 单调，且有界 $a,b$，故 $\{x_n\}$ 收敛于某点 $x_0$. 在 $g(x_n)=f(x_{n+1})$ 中取极限，由 $f,g$ 的连续性得 $g(x_0)=f(x_0)$.

\noindent\textbf{注}\quad 有界无穷数列必有收敛子列，因此在 $\{x_n\}$ 中能找出收敛子列 $x_{n_k}\to x_0$（某极限点），似乎可以删去 $f$ 的单调性条件. 但这种方法得到的子列，其邻项不能保证具有关系：$g(x_{n_k})=f(x_{n_k+1})$.

\noindent\textbf{*2.1.23}\quad 设 $f(x)$ 在 $(a,+\infty)$ 内连续、有界，试证：$\forall T,\exists x_n\to+\infty$ 使得
\[
  \lim_{n\to\infty}[f(x_n+T)-f(x_n)]=0.
\]

\noindent\textbf{证}\quad （记 $g(x)=f(x+T)-f(x)$，来找 $x_n\to+\infty$ 使得 $g(x_n)\to0$.）事实上，若 $x\nearrow+\infty$ 时，$g(x)$ 无穷次变号，则问题明显. 只需证明 $x$ 充分大后 $g(x)$ 保持不变号（例如恒有 $g(x)>0$）的情况即可.

由 $f$ 的有界性知：$\forall\varepsilon>0,x$ 充分大之后，不能永远有 $g(x)\ge\varepsilon$（不然，$f(x+(n+1)T)=\sum\limits_{k=0}^{n}g(x+kT)+f(x)\ge(n+1)\varepsilon+f(x)\to+\infty$（当 $n\to\infty$ 时）与 $f$ 有界矛盾）. 因此，对充分大的每个自然数 $n$，$\exists x_n>n$，使得 $g(x_n)<\dfrac1n$. 证毕.

\noindent\textbf{**2.1.24}\quad 设 $f$ 在 $[0,n]$ 上连续（$n$ 为自然数），$f(0)=f(n)$. 试证：至少存在 $n$ 组不同的解 $(x,y)$ 使得 $f(x)=f(y)$，且 $y-x>0$ 为整数.

\noindent\textbf{证}\quad （用数学归纳法.）$1^\circ$ $n=1$ 时明显.

$2^\circ$（由 $n=k$ 时成立推出 $n=k+1$ 时成立.）考虑函数 $g(x)=f(x+1)-f(x)$，当 $n=k+1$ 时，有 $f(0)=f(k+1)$，故 $\sum\limits_{i=0}^{k}g(i)=0$，可知总 $\exists\xi\in[0,k]$，使得 $g(\xi)=0$，即 $f(\xi+1)=f(\xi)$，$(\xi,\xi+1)$ 为一组解.

作函数
\[
  \varphi(x)=
  \begin{cases}
    f(x), & x\in[0,\xi],\\
    f(x+1), & x\in(\xi,k]
  \end{cases}
\quad(\text{平移相接}).
\]
这时 $\varphi(0)=\varphi(k)$ 满足命题对 $n=k$ 的条件，应用到 $\varphi$ 可得 $k$ 组解. 但 $\varphi$ 的解也必是 $f$ 的解，且与 $(\xi,\xi+1)$ 不同，如此获得 $f$ 的 $n+1$ 组解. 得证.

\noindent\textbf{*2.1.25}\quad 用确界原理（非空有上（下）界数集必有上（下）确界）证明：若 $f(x)$ 在 $[a,b]$ 上连续，$f(a)\cdot f(b)<0$，则存在一点 $c\in(a,b)$，使 $f(c)=0$.（西北大学）

\noindent\textbf{提示}\quad 不妨设 $f(a)<0,f(b)>0$，于是 $A=\{x\mid f(x)<0\}$ 非空，有上界 $b$. 由确界原理，$\exists c\in[a,b]:c=\sup\{x\mid f(x)<0\}\Rightarrow \forall\dfrac1n>0,\exists x_n\in A:c-\dfrac1n<x_n<c$（$n=1,2,\cdots$）
\[
  \overset{f\text{ 连续}}{\Longrightarrow}\lim_{n\to\infty}f(x_n)=f(c)\le0\quad(\text{因 }f(x_n)<0).
\]
又 $c+\dfrac1n\notin A,f\left(c+\dfrac1n\right)\ge0$，$f(c)=\lim\limits_{n\to\infty}f\left(c+\dfrac1n\right)\ge0$，故 $f(c)=0,c\in(a,b)$.

\noindent\textbf{*2.1.26}\quad 设 $f(x)$ 在 $[a,b]$ 上连续，$f(a)\cdot f(b)<0$，应用闭区间套原理证明：至少存在一点 $\xi\in(a,b)$，使得 $f(\xi)=0$.（北京科技大学）

\noindent\textbf{提示}\quad 将 $[a,b]$ 二等分，若中点处 $f\left(\dfrac{a+b}{2}\right)\ne0$，两子区间，必有一个 $f$ 在端点异号；将其再二等分；如此下去组成闭区间套 $\{[a_n,b_n]\}$，其公共点 $\xi\in[a_n,b_n]$（$n=1,2,\cdots$），必有 $f(\xi)=0$.

\noindent\textbf{*2.1.27}\quad 用有限覆盖定理证明连续函数的零点定理：若 $f(x)$ 在 $[a,b]$ 上连续，$f(a)\cdot f(b)<0$，则存在 $\xi\in(a,b)$，使得 $f(\xi)=0$.（四川大学）

\noindent\textbf{提示}\quad （反证法）否则，$\forall x\in[a,b],f(x)\ne0\Rightarrow f(x)>0$（或 $<0$）$\overset{\text{连续保号性}}{\Longrightarrow}\exists\delta_x>0,(x-\delta_x,x+\delta_x)\cap[a,b]$ 内 $f(x)$ 保持同号；$\{(x-\delta_x,x+\delta_x)\mid x\in[a,b]\}$ 组成 $[a,b]$ 的开覆盖 $\Rightarrow$ 其中存在有限子覆盖，相邻接的两开区间必有公共点 $\Rightarrow$ 相邻区间里 $f$ 同号 $\Rightarrow f(a)$ 与 $f(b)$ 同号（矛盾）.

\noindent\textbf{*2.1.28}\quad 用闭区间套定理证明连续函数有界性定理，即若 $f(x)$ 在闭区间 $[a,b]$ 上连续，则存在 $M>0$，对一切 $x\in[a,b]$，$|f(x)|\le M$.（华中师范大学）

\noindent\textbf{new 2.1.29}\quad 1）用闭区间套定理证明有限覆盖定理（注：基本定理互证，这种考研题多见）；

2）试用有限覆盖定理证明：$[a,b]$ 上的正值连续函数 $f(x)$ 必有正的下界 $m$：
\[
  f(x)\ge m>0\quad(x\in[a,b]).
\]
（浙江大学）

\noindent\textbf{提示}\quad 1）参看例 1.8.4.

2）（因 $f(x)>0$，连续函数有保号性.）对每个 $x_i\in[a,b]$，$\exists\delta_i>0$，使得当 $[a,b]$ 中的 $x\in(x_i-\delta_i,x_i+\delta_i)$ 时，就有 $f(x)\ge\dfrac{f(x_i)}2>0$. 于是，开区间组成的集合 $\{(x_i-\delta_i,x_i+\delta_i)\}_{x_i\in[a,b]}$ 构成了 $[a,b]$ 的一个覆盖. 根据有限覆盖定理，从中能挑出有限子覆盖（记为）$(x_i-\delta_i,x_i+\delta_i),i=1,2,\cdots,N$. 如此，$\forall x\in[a,b]\subset\bigcup\limits_{i=1}^{N}(x_i-\delta_i,x_i+\delta_i)$，有
\[
  f(x)\ge\min_{1\le i\le N}\left\{\frac{f(x_i)}2\right\}\overset{\text{记}}=m>0.
\]

\section*{* §2.2\quad 一致连续性}

\noindent\textbf{导读}\quad 一致连续性是数学院系学生的重点内容，非数学院系学生不作要求. 本节主要讨论如何利用一致连续性的定义及其否定形式，来证明函数一致连续与非一致连续. 其次，讨论一致连续与连续的关系. 最后介绍连续模数及一致连续函数的延拓问题.

关于上、下半连续的内容，本书只对基本概念和结论作一综述，有需要的读者可参考本书第 2 版.

\subsection*{一、利用一致连续的定义及其否定形式证题}

\noindent\textbf{要点}\quad 设 $f(x)$ 在区间 $I$ 上有定义（$I$ 为开、闭、半开半闭，有限或无限区间）. 所谓 $f(x)$ 在 $I$ 上一致连续，意指：$\forall\varepsilon>0,\exists\delta>0$，当 $x',x''\in I,|x'-x''|<\delta$ 时，有
\[
  |f(x')-f(x'')|<\varepsilon.
\]
如此，
\[
\begin{aligned}
&f(x)\text{ 在 }I\text{ 上非一致连续}\\
&\Leftrightarrow \exists\varepsilon_0>0,\forall\delta>0,\exists x_\delta',x_\delta''\in I:
  \text{虽然 }|x_\delta'-x_\delta''|<\delta,\text{但是 }|f(x_\delta')-f(x_\delta'')|\ge\varepsilon_0\\
&\Leftrightarrow \exists\varepsilon_0>0,\forall\frac1n>0,\exists x_n',x_n''\in I\ (n=1,2,\cdots):
  \text{虽然 }|x_n'-x_n''|<\frac1n,\text{但是 }|f(x_n')-f(x_n'')|\ge\varepsilon_0.
\end{aligned}
\]
特别，若 $\exists\varepsilon_0>0,\exists x_n',x_n''\in I$（$n=1,2,\cdots$），虽然
\[
  \lim_{n\to\infty}x_n'=\lim_{n\to\infty}x_n''=a,
\]
但是
\[
  |f(x_n')-f(x_n'')|\ge\varepsilon_0\quad(n=1,2,\cdots),
\]
则可断定 $f$ 在 $I$ 上非一致连续.

用定义证明 $f$ 在 $I$ 上一致连续，通常的方法是设法证明 $f$ 在 $I$ 满足 Lipschitz 条件：
\[
  |f(x')-f(x'')|\le L|x'-x''|,\quad \forall x',x''\in I,
\]
其中 $L>0$ 为某一常数. 此条件成立必一致连续.

特别，若 $f$ 在 $I$ 上存在有界导函数，则 $f$ 在 $I$ 满足 Lipschitz 条件.

\noindent\textbf{例 2.2.1}\quad 设 $f(x)=\dfrac{x+2}{x+1}\sin\dfrac1x$，$a>0$ 为任一正常数，试证：$f(x)$ 在 $(0,a)$ 内非一致连续，在 $[a,+\infty)$ 上一致连续.（兰州大学）

\noindent\textbf{证}\quad $1^\circ$ 证明 $f$ 在 $[a,+\infty)$ 上一致连续.

\noindent\textbf{方法 I}\quad 证明 $f$ 在 $[a,+\infty)$ 上满足 Lipschitz 条件. $\forall x',x''\in[a,+\infty)$，
\[
\begin{aligned}
|f(x')-f(x'')|
&\le \left|\frac{x'+2}{x'+1}\sin\frac1{x'}-\frac{x''+2}{x''+1}\sin\frac1{x'}\right|
 +\left|\frac{x''+2}{x''+1}\sin\frac1{x'}-\frac{x''+2}{x''+1}\sin\frac1{x''}\right|\\
&\le \left|\frac{x'+2}{x'+1}-\frac{x''+2}{x''+1}\right|
 +\frac{x''+2}{x''+1}\cdot2
 \left|\cos\frac{\frac1{x'}+\frac1{x''}}2\right|
 \left|\sin\frac{\frac1{x'}-\frac1{x''}}2\right|\\
&\le \frac{|x''-x'|}{(x'+1)(x''+1)}
 +\left(1+\frac1{x''+1}\right)\cdot2\cdot\frac{\left|\frac1{x'}-\frac1{x''}\right|}{2}\\
&\le \left[\frac1{(a+1)^2}+\frac{a+2}{a^2(a+1)}\right]|x'-x''|
 =L|x'-x''|.
\end{aligned}
\]
从而 $\forall\varepsilon>0,\exists\delta=\dfrac\varepsilon L$，当 $|x'-x''|<\dfrac\varepsilon L$ 时，$|f(x')-f(x'')|<\varepsilon$.

\noindent\textbf{方法 II}\quad 证明 $f'(x)$ 在 $[a,+\infty)$ 上有界. 略.

$2^\circ$ 证明 $f$ 在 $(0,a)$ 内非一致连续.

取
\[
  x_n'=\frac1{2n\pi+\frac\pi2},\quad
  x_n''=\frac1{2n\pi-\frac\pi2}\quad(n=1,2,\cdots),
\]
则 $n$ 充分大时，$x_n',x_n''\in(0,a)$，且
\[
  |x_n'-x_n''|=\frac{\pi}{4n^2\pi^2-\frac{\pi^2}{4}}\to0\quad(\text{当 }n\to\infty\text{ 时}).
\]
但
\[
  |f(x_n')-f(x_n'')|
  =\left|\frac{4n\pi+\pi+1}{2n\pi+\frac\pi2+1}
  +\frac{4n\pi-\pi+1}{2n\pi-\frac\pi2+1}\right|>2.
\]
故 $f$ 在 $(0,a)$ 内非一致连续.

\noindent\textbf{例 2.2.2}\quad 证明：$f(x)=\dfrac1x$ 在 $(0,1)$，$g(x)=x^2$ 在 $(1,+\infty)$ 内非一致连续.

\noindent\textbf{证}\quad
\[
  \left|\frac1n-\frac1{n+1}\right|\to0\quad(n\to\infty),
  \quad\text{但}\quad
  \left|\frac1{\frac1n}-\frac1{\frac1{n+1}}\right|\equiv1.
\]
\[
  |\sqrt{n+1}-\sqrt n|=\frac1{\sqrt{n+1}+\sqrt n}\to0,
\]
但
\[
  \left|(\sqrt{n+1})^2-(\sqrt n)^2\right|=1.
\]
故 $f$ 在 $(0,1)$，$g$ 在 $(1,+\infty)$ 内都非一致连续.

\noindent\textbf{$\star$例 2.2.3}\quad 证明：$f(x)$ 在区间 $I$ 上一致连续的充要条件是：对 $I$ 上任意两数列 $\{x_n\},\{x_n'\}$，只要 $x_n-x_n'\to0$，就有 $f(x_n)-f(x_n')\to0(n\to\infty)$.（华中科技大学）

\noindent\textbf{证}\quad $1^\circ$（必要性）因 $f$ 一致连续，所以 $\forall\varepsilon>0,\exists\delta>0$，当 $x,x'\in I,|x-x'|<\delta$ 时，有
\[
  |f(x)-f(x')|<\varepsilon. \tag{1}
\]
但 $x_n-x_n'\to0$（当 $n\to\infty$ 时），故对 $\delta>0,\exists N>0$，当 $n>N$ 时，$|x_n-x_n'|<\delta$，从而由式 (1)，$|f(x_n)-f(x_n')|<\varepsilon$. 即
\[
  f(x_n)-f(x_n')\to0\quad(\text{当 }n\to\infty\text{ 时}).
\]

$2^\circ$（充分性）若 $f$ 在 $I$ 上非一致连续，则 $\exists\varepsilon_0>0,\forall\dfrac1n>0,\exists x_n,x_n'\in I$，虽然
\[
  |x_n-x_n'|<\frac1n,\text{ 但 } |f(x_n)-f(x_n')|\ge\varepsilon_0.
\]
可见 $x_n-x_n'\to0$，但 $f(x_n)-f(x_n')\not\to0$（当 $n\to\infty$ 时），矛盾.

\noindent\textbf{$\star$例 2.2.4}\quad 设 $I$ 为有限区间，$f(x)$ 在 $I$ 上有定义，试证：$f(x)$ 在 $I$ 上一致连续的充要条件是 $f$ 把 Cauchy 序列映射为 Cauchy 序列\footnote{Cauchy 序列又称基本序列，或自收敛序列，意指所讨论的序列 $\{a_n\}$ 满足 Cauchy 条件：$\forall\varepsilon>0,\exists N>0,m,n>N$ 时 $|a_m-a_n|<\varepsilon$.}（即当 $\{x_n\}$ 为 Cauchy 序列时，$\{f(x_n)\}$ 亦为 Cauchy 序列）.（北京师范大学，西北师范大学，河南师范大学）

\noindent\textbf{证}\quad $1^\circ$（必要性）已知 $\forall\varepsilon>0,\exists\delta>0$，当 $x',x''\in I,|x'-x''|<\delta$ 时，有
\[
  |f(x')-f(x'')|<\varepsilon. \tag{1}
\]
设 $\{x_n\}$ 为 Cauchy 序列，则对此 $\delta>0,\exists N>0$，当 $n,m>N$ 时，有 $|x_n-x_m|<\delta$，从而由式 (1)，
\[
  |f(x_n)-f(x_m)|<\varepsilon,
\]
所以 $\{f(x_n)\}$ 亦为 Cauchy 序列.

$2^\circ$（充分性）若 $f(x)$ 在 $I$ 上非一致连续，则 $\exists\varepsilon_0>0,\forall\delta_n=\dfrac1n>0,\exists x_n,x_n'\in I$：虽然
\[
  |x_n-x_n'|<\frac1n,\text{ 但 }
\]
\[
  |f(x_n)-f(x_n')|\ge\varepsilon_0\quad(n=1,2,\cdots). \tag{2}
\]
注意到 $I$ 为有限区间，$x_n\in I(n=1,2,\cdots)$，因此 $\{x_n\}$ 中存在收敛子列 $\{x_{n_k}\}$. 因为 $|x_n-x_n'|\to0$（当 $n\to\infty$ 时），故 $\{x_{n_k}'\}$ 中相应的子列 $\{x_{n_k}'\}$ 也收敛于相同的极限. 从而穿插之后，序列
\[
  x_{n_1},x_{n_1}',x_{n_2},x_{n_2}',\cdots,x_{n_k},x_{n_k}',\cdots
\]
亦收敛，为 Cauchy 序列. 但其像序列
\[
  f(x_{n_1}),f(x_{n_1}'),f(x_{n_2}),f(x_{n_2}'),\cdots,f(x_{n_k}),f(x_{n_k}'),\cdots
\]
恒有
\[
  |f(x_{n_k})-f(x_{n_k}')|\ge\varepsilon_0,
\]
不是 Cauchy 序列，与已知条件矛盾.

\noindent\textbf{注}\quad $I$ 的有限性只在充分性用到. 对无穷区间，必要性仍成立.

\noindent\textbf{例 2.2.5}\quad 设 $z=g(y)$ 于 $J,y=f(x)$ 于 $I$ 都是一致连续的，且 $f(I)\subset J$. 试证 $z=g(f(x))$ 在 $I$ 上一致连续.

\noindent\textbf{提示}\quad 可直接利用定义；当 $I,J$ 有限时可利用上例.

\subsection*{二、一致连续与连续的关系}

我们知道 $f(x)$ 在区间 $I$ 上一致连续，自然 $f(x)$ 在 $I$ 上连续，反之不一定. 若 $I$ 为有限闭区间，根据 Cantor 定理，$f$ 在 $[a,b]$ 上连续等价于 $f$ 在 $[a,b]$ 上一致连续.

现在让我们来讨论开区间以及无穷区间的情况.

\noindent\textbf{$\star$例 2.2.6}\quad 设 $f(x)$ 在有限开区间 $(a,b)$ 内连续，试证 $f(x)$ 在 $(a,b)$ 内一致连续的充要条件是极限 $\lim\limits_{x\to a+}f(x)$ 及 $\lim\limits_{x\to b-}f(x)$ 存在（有限）.（山东大学，南开大学）

\noindent\textbf{证}\quad $1^\circ$（必要性）已知 $\forall\varepsilon>0,\exists\delta>0$，当 $x',x''\in(a,b),|x'-x''|<\delta$ 时，有
\[
  |f(x')-f(x'')|<\varepsilon.
\]
故 $\forall x',x''\in(a,b),a<x'<a+\delta,a<x''<a+\delta$ 时，有
\[
  |f(x')-f(x'')|<\varepsilon.
\]
根据 Cauchy 准则，知 $\lim\limits_{x\to a+}f(x)$ 存在（有限）. 同理 $\lim\limits_{x\to b-}f(x)$ 存在.

$2^\circ$（充分性）补充定义 $f(a)=\lim\limits_{x\to a+}f(x),f(b)=\lim\limits_{x\to b-}f(x)$，则 $f(x)$ 在 $[a,b]$ 上连续. 由 Cantor 定理，$f(x)$ 在 $[a,b]$ 上一致连续. 从而 $f(x)$ 在 $(a,b)$ 内一致连续.

\noindent\textbf{注}\quad （1）此例表明：在有限开区间上连续函数是否一致连续，取决于函数在端点附近的状态. 应用本例，容易判明 $y=\dfrac1x\sin x$ 在 $(0,1)$ 内一致连续；而 $y=\sin\dfrac1x,y=\ln x,y=\dfrac1{1-x}$ 在 $(0,1)$ 内非一致连续.

（2）由此例还可看出，$f(x)$ 在 $(a,b)$ 内一致连续，则 $f$ 在 $(a,b)$ 内有界. 然而，在开区间上连续、有界，不一定一致连续，如 $y=\sin\dfrac1x$.

（3）当 $(a,b)$ 改为无穷区间时，该例的必要性不再成立. 如 $f(x)=x,g(x)=\sin x$ 在 $(-\infty,+\infty)$ 上一致连续，但在端点 $\pm\infty$ 无极限. 对于无穷区间，充分性仍是对的. 请看

\noindent\textbf{$\star$例 2.2.7}\quad 证明：若 $f(x)$ 在 $[a,+\infty)$ 上连续，$\lim\limits_{x\to+\infty}f(x)=A$（有限），则 $f(x)$ 在 $[a,+\infty)$ 上一致连续.（新疆大学，中国人民大学）

\noindent\textbf{证}\quad $1^\circ$ 因 $\lim\limits_{x\to+\infty}f(x)=A$，所以 $\forall\varepsilon>0,\exists\Delta>a$，当 $x',x''>\Delta$ 时，有
\[
  |f(x')-f(x'')|<\varepsilon \tag{1}
\]
（Cauchy 准则之“必要性”）.

$2^\circ$ 由 Cantor 定理，$f$ 在 $[a,\Delta+1]$ 上一致连续，故对此 $\varepsilon>0,\exists\delta_1>0$，当 $x',x''\in[a,\Delta+1],|x'-x''|<\delta_1$ 时，有
\[
  |f(x')-f(x'')|<\varepsilon. \tag{2}
\]

$3^\circ$ 令 $\delta=\min\{1,\delta_1\}$，则 $x',x''>a,|x'-x''|<\delta$ 时，$x',x''$ 要么同属于 $[a,\Delta+1]$，要么同属于 $(\Delta,+\infty)$. 从而由式 (1)、(2) 知 $|f(x')-f(x'')|<\varepsilon$，即 $f$ 在 $[a,+\infty)$ 上一致连续.

\noindent\textbf{注}\quad 如下证明是错误的：首先利用以上证明的 $1^\circ$，得结论“$f$ 在 $[\Delta,+\infty)$ 上一致连续”，然后利用 Cantor 定理，$f$ 在 $[a,\Delta]$ 上一致连续，从而 $f$ 在 $[a,+\infty)$ 上一致连续. 其错误在于 $1^\circ$ 中 $\Delta$ 与 $\varepsilon$ 有关，由 $1^\circ$ 得不出 $f$ 在 $[\Delta,+\infty)$ 上一致连续.

\noindent\textbf{$\star$例 2.2.8}\quad 设 $f(x)$ 在 $[a,+\infty)$ 上一致连续，$\varphi(x)$ 在 $[a,+\infty)$ 上连续，$\lim\limits_{x\to+\infty}[f(x)-\varphi(x)]=0$. 证明：$\varphi(x)$ 在 $[a,+\infty)$ 上一致连续.（上海交通大学，华中科技大学）

\noindent\textbf{证}\quad $1^\circ$ 因 $\lim\limits_{x\to+\infty}[f(x)-\varphi(x)]=0$，所以 $\forall\varepsilon>0,\exists\Delta>a$，当 $x>\Delta$ 时，$|f(x)-\varphi(x)|<\dfrac\varepsilon3$. 又因 $f$ 一致连续，故对此 $\varepsilon>0,\exists\delta_1>0$，当 $|x'-x''|<\delta_1$ 时 $|f(x')-f(x'')|<\dfrac\varepsilon3$. 如此，$\forall x',x''>\Delta,|x'-x''|<\delta_1$ 时有
\[
\begin{aligned}
|\varphi(x')-\varphi(x'')|
&\le |\varphi(x')-f(x')|+|f(x')-f(x'')|+|f(x'')-\varphi(x'')|\\
&<\frac\varepsilon3+\frac\varepsilon3+\frac\varepsilon3=\varepsilon.
\end{aligned}
\]

$2^\circ$ 利用 Cantor 定理，可知 $\varphi(x)$ 在 $[a,\Delta+1]$ 上一致连续，所以对此 $\varepsilon>0,\exists\delta_2>0$，当 $x',x''\in[a,\Delta+1],|x'-x''|<\delta_2$ 时，有 $|\varphi(x')-\varphi(x'')|<\varepsilon$.

$3^\circ$ 取 $\delta=\min\{1,\delta_1,\delta_2\}$，则 $x',x''\in[a,+\infty),|x'-x''|<\delta$ 时，有 $|\varphi(x')-\varphi(x'')|<\varepsilon$. 证毕.

我们知道，$y=x$ 在 $(-\infty,+\infty)$ 上一致连续，但 $y=x^2$ 在 $(-\infty,+\infty)$ 上非一致连续\footnote{如令 $x_n=\sqrt{n+1},x_n'=\sqrt n$，这时 $|x_n-x_n'|=\dfrac1{\sqrt{n+1}+\sqrt n}\to0$，但 $|x_n^2-x_n'^2|=|n+1-n|=1$，可见 $y=x^2$ 在 $(-\infty,+\infty)$ 上不一致连续.}. 我们要问：在无穷区间上一致连续的函数，当 $x\to\pm\infty$ 时，阶次有何估计？

\noindent\textbf{*例 2.2.9}\quad 设 $f(x)$ 在 $(-\infty,+\infty)$ 上一致连续，则存在非负实数 $a$ 与 $b$，使对一切 $x\in(-\infty,+\infty)$，都有
\[
  |f(x)|\le a|x|+b.
\]
试证明之.（云南大学，南开大学）

\noindent\textbf{证}\quad 因为 $f(x)$ 一致连续，所以 $\forall\varepsilon>0,\exists\delta>0$，当 $|x'-x''|\le\delta$ 时，有 $|f(x')-f(x'')|<\varepsilon$. 现将 $\varepsilon>0,\delta>0$ 固定. 由于 $\forall x\in(-\infty,+\infty),\exists n\in\mathbb{Z}$（整数集），使得 $x=n\delta+x_0$，其中 $x_0\in(-\delta,\delta)$. 注意到 $f(x)$ 在 $[-\delta,\delta]$ 上有界，即 $\exists M>0$，使得 $|f(x)|\le M(\forall x\in[-\delta,\delta])$. 因此
\[
  f(x)=\sum_{k=1}^{n}\{f(k\delta+x_0)-f[(k-1)\delta+x_0]\}+f(x_0),
\]
\[
  |f(x)|\le\sum_{k=1}^{n}|f(k\delta+x_0)-f[(k-1)\delta+x_0]|+|f(x_0)|\le |n|\varepsilon+M.
\]
由 $x=n\delta+x_0$ 知 $\left|\dfrac{x-x_0}{\delta}\right|=|n|$，代入上式得
\[
  |f(x)|\le\frac\varepsilon\delta|x-x_0|+M\le\frac\varepsilon\delta|x|+\left(M+\frac\varepsilon\delta|x_0|\right)\le\frac\varepsilon\delta|x|+(M+\varepsilon).
\]
记 $\dfrac\varepsilon\delta=a,M+\varepsilon=b$，则 $a>0,b>0$，
\[
  |f(x)|\le a|x|+b\quad(\forall x\in(-\infty,+\infty)).
\]
此例说明，若 $f(x)$ 在 $(-\infty,+\infty)$ 上一致连续，则 $x\to\infty$ 时，$f(x)=O(x)$.

下面我们来看一个使用一致连续性的例子.

\noindent\textbf{例 2.2.10}\quad 设函数 $f(x)$ 在 $[0,+\infty)$ 上一致连续，且 $\forall x\ge0$ 有 $\lim\limits_{n\to\infty}f(x+n)=0$（$n$ 为正整数）. 试证 $\lim\limits_{x\to+\infty}f(x)=0$.（南昌大学，上海师范大学）

\noindent\textbf{分析}\quad 要证明 $\lim\limits_{x\to+\infty}f(x)=0$，即要证明：$\forall\varepsilon>0\ \exists\Delta>0$，当 $x>\Delta$ 时，有 $|f(x)|<\varepsilon$.

已知 $\forall x>0$，有 $\lim\limits_{n\to\infty}f(x+n)=0$. 因此对区间 $[0,1]$ 上的每个点 $x\in[0,1]$，相应 $\exists N_x>0$，当 $n>N_x$ 时 $|f(x+n)|<\varepsilon$. 可惜这样找得的 $N_x(x\in[0,1])$ 共有无穷多个，无法从中找出最大的 $N$. 为此，将 $[0,1]$ $k$ 等分，对每个分点 $x_i=\dfrac ik(i=1,2,\cdots,k)$，相应 $\exists N_i>0$，使得当 $n>N_i$ 时，$|f(x_i+n)|<\varepsilon$. 令 $N=\max\{N_1,N_2,\cdots,N_k\}$，则当 $n>N$ 时，有 $|f(x_i+n)|<\varepsilon(i=1,2,\cdots,k)$. 如此我们虽未找到所需的 $\Delta>0$，但至少在 $[N,+\infty)$ 内的每个格点 $x_i+n(i=1,2,\cdots,k,n=N+1,N+2,\cdots)$ 上，有 $|f(x_i+n)|<\varepsilon$. 注意到 $f(x)$ 在 $[0,+\infty)$ 上一致连续，因此把分划取得足够细，使得格点足够密，可使两格点之间的函数值，与格点的函数值相差任意小.

\noindent\textbf{证}\quad $1^\circ$ 因 $f(x)$ 在 $[0,+\infty)$ 上一致连续，故 $\forall\varepsilon>0,\exists\delta>0$，当 $|x'-x''|<\delta(x',x''>0)$ 时，有
\[
  |f(x')-f(x'')|<\frac\varepsilon2. \tag{1}
\]

$2^\circ$ 取 $k>\dfrac1\delta$，将区间 $[0,1]$ $k$ 等分. 记分点 $x_i=\dfrac ik(i=1,2,\cdots,k)$，这时间距 $x_i-x_{i-1}=\dfrac1k<\delta$.

$3^\circ$ 由已知条件，对每个 $x_i=\dfrac ik$，有 $\lim\limits_{n\to\infty}f(x_i+n)=0$. 从而 $\exists N_i>0$，使得 $n>N_i$ 时，有 $|f(x_i+n)|<\dfrac\varepsilon2$. 令 $N=\max\limits_{1\le i\le k}\{N_i\}$，则当 $n>N$ 时，
\[
  |f(x_i+n)|<\frac\varepsilon2\quad(i=1,2,\cdots,k). \tag{2}
\]

$4^\circ$ 取 $\Delta=N>0$，来证明当 $x>\Delta$ 时 $|f(x)|<\varepsilon$. 事实上，$\forall x>N$，记 $n=[x]$\footnote{$[x]$ 表示不大于 $x$ 的最大整数.}$\ge N$，因 $x-n\in[0,1]$，故 $\exists i\in\{1,2,\cdots,k\}$，使得 $|(x-n)-x_i|<\delta$，即 $|x-(n+x_i)|<\delta$，由式 (1)，$|f(x)-f(n+x_i)|<\dfrac\varepsilon2$. 再由式 (2)，
\[
  |f(x)|\le |f(x)-f(n+x_i)|+|f(n+x_i)|<\frac\varepsilon2+\frac\varepsilon2=\varepsilon.
\]
即 $\lim\limits_{x\to+\infty}f(x)=0$.

由此例易知：若 $g$ 在 $[a,+\infty)$ 上一致连续，且 $\forall x\ge0$，有 $\lim\limits_{n\to\infty}g(x+n)=A$，则
\[
  \lim_{x\to+\infty}g(x)=A.
\]

\subsection*{三、用连续模数描述一致连续性}

\noindent\textbf{定义 1}\quad 设 $f(x)$ 在区间 $I$ 上有定义，
\[
  \omega_f(\delta)=\sup_{\substack{x',x''\in I\\ |x'-x''|\le\delta}}|f(x')-f(x'')|
\]
称为函数 $f$ 的连续模数.

可见 $\omega_f(\delta)$ 是关于 $\delta$ 的非负、不减函数. 下面我们借助它来描述一致连续性.

\noindent\textbf{*例 2.2.11}\quad 若 $f(x)$ 在区间 $I$ 上有定义，则 $f(x)$ 在 $I$ 上一致连续的充要条件是
\[
  \lim_{\delta\to0+}\omega_f(\delta)=0.
\]

\noindent\textbf{证}\quad $1^\circ$ 必要性. 因 $f(x)$ 在 $I$ 上一致连续，因此 $\forall\varepsilon>0,\exists\delta_1>0$，当 $x',x''\in I,|x'-x''|<\delta_1$ 时，有 $|f(x')-f(x'')|<\dfrac\varepsilon2$. 从而
\[
  \omega_f(\delta_1)=\sup_{\substack{x',x''\in I\\ |x'-x''|\le\delta_1}}|f(x')-f(x'')|\le\frac\varepsilon2.
\]
故 $0<\delta<\delta_1$ 时，$0\le\omega_f(\delta)\le\omega_f(\delta_1)\le\dfrac\varepsilon2<\varepsilon$. 所以 $\lim\limits_{\delta\to0+}\omega_f(\delta)=0$.

$2^\circ$ 充分性. 由 $\lim\limits_{\delta\to0+}\omega_f(\delta)=0$ 知：$\forall\varepsilon>0,\exists\delta_1>0$ 使得 $0\le\omega_f(\delta_1)<\varepsilon$，故当 $x',x''\in I,|x'-x''|<\delta_1$ 时，有
\[
  |f(x')-f(x'')|\le
  \sup_{\substack{x',x''\in I\\ |x'-x''|\le\delta_1}}|f(x')-f(x'')|
  =\omega_f(\delta_1)<\varepsilon.
\]
所以 $f$ 在 $I$ 上一致连续.

\noindent\textbf{注}\quad 由此可得一致连续的观察法：因为 $\omega_f(\delta)$ 的值只与 $f$ 的图形最陡的地方有关. 若 $f$ 的图形在某处无限变陡，使得 $\omega_f(\delta)\not\to0(\delta\to0)$，则 $f$ 非一致连续；若 $f$ 在某处最陡，但当 $\delta\to0+$ 时，此处的变差 $|f(x')-f(x'')|\to0$，则 $f$ 一致连续.

例如 $f(x)=\dfrac1x(x>0)$，在 $x=0$ 处图形无限变陡. $\forall\delta>0,\omega_f(\delta)=+\infty,\delta\to0+$ 时 $\omega_f(\delta)\not\to0$. 因此 $f$ 在任何区间 $(0,c)(c>0)$ 内都是非一致连续的. 但在区间 $[c,+\infty)$ 上，$f(x)=\dfrac1x$ 在点 $c$ 处最陡，且 $\omega_f(\delta)=\dfrac1c-\dfrac1{c+\delta}\to0$（当 $\delta\to0+$ 时）. 可见 $f(x)=\dfrac1x$ 在 $[c,+\infty)$ 上一致连续.

类似，我们容易看出 $\ln x,e^x\cos\dfrac1x,\arctan x,\sin\dfrac1x$ 的一致连续区间，留给读者考虑.

\subsection*{※四、集上的连续函数及一致连续函数的延拓问题}

前面例 2.2.6 告诉我们，若 $f(x)$ 在 $(a,b)$ 内一致连续，则 $f$ 在端点 $a,b$ 处有有限极限，因此若将极限值分别作为 $f$ 在 $a,b$ 点的值，那么 $f$ 被延拓到闭区间 $[a,b]$ 上，且在 $[a,b]$ 上一致连续，下面我们将此结论推广到一般的集合 $E\subset\mathbb{R}$ 上. 为此，我们首先把连续与一致连续的概念推广到任意的集合 $E$ 上.

\noindent\textbf{定义 2}\quad $f(x)$ 在集合 $E\subset\mathbb{R}$ 上有定义，$x_0\in E$，所谓 $f(x)$ 在 $x_0$ 处连续，指：$\forall\varepsilon>0,\exists\delta>0$，当 $|x-x_0|<\delta,x\in E$ 时，有 $|f(x)-f(x_0)|<\varepsilon$. 若 $f$ 在 $E$ 的每点上都连续，则称 $f$ 在 $E$ 上连续.

换句话说，定义与区间上连续的定义一样，只是把区间 $I$ 改为了集合 $E$. 类似可定义 $E$ 上的一致连续性.

值得注意的是，按此定义，$f$ 在 $E$ 的孤立点（如果存在）$x_0$ 必然连续. 因 $\delta>0$ 取得充分小时，$x_0$ 的 $\delta$ 邻域 $(x_0-\delta,x_0+\delta)$ 与 $E$ 只有一个公共点 $x_0$. 可见今后只需讨论 $f$ 在 $E$ 的聚点处的连续性.

按此定义，易知 Dirichlet 函数
\[
  D(x)=
  \begin{cases}
    1, & \text{当 }x\text{ 为有理数时},\\
    0, & \text{当 }x\text{ 为无理数时}
  \end{cases}
\]
处处不连续. 但限制在集合 $\mathbb{Q}=\{\text{有理数}\}$ 上考虑，则 $D(x)$ 在 $\mathbb{Q}$ 上连续.

下面讨论延拓问题.

\noindent\textbf{**例 2.2.12}\quad 设 $E$ 为实轴 $\mathbb{R}$ 上的一个集合，$E_1\subset E$ 为 $E$ 的稠密子集（即 $\forall x\in E,\exists x_n\in E_1(n=1,2,\cdots)$，使得 $x_n\to x$（当 $n\to\infty$ 时））. 若 $f_1(x)$ 是 $E_1$ 上的一致连续函数（即：$\forall\varepsilon>0,\exists\delta>0$，当 $x',x''\in E_1,|x'-x''|<\delta$ 时，有 $|f_1(x')-f_1(x'')|<\varepsilon$），则在 $E$ 上有唯一函数 $f(x)$ 使得

1）$f(x)=f_1(x)$（当 $x\in E_1$ 时）；

2）$f(x)$ 在 $E$ 上连续. 特别 $E$ 为有界集合时，$f(x)$ 在 $E$ 上一致连续.

\noindent\textbf{证}\quad 证明的思路如下：已知对任意 $x_0\in E$，可取出序列 $\{x_n\}\subset E_1$，使得 $x_n\to x_0$. 我们来证明 $\lim\limits_{n\to\infty}f_1(x_n)$ 存在，且极限值与 $\{x_n\}$ 的取法无关，函数 $f(x)$：
\[
  f(x_0)=\lim_{n\to\infty}f_1(x_n)
\]
在 $E$ 上被任意确定. 然后证明此函数在 $E$ 上满足条件 1）和 2），并且符合此条件的函数是唯一的.

具体地说：

$1^\circ$ 已知 $\forall x_0\in E,\exists x_n\in E_1(n=1,2,\cdots)$，使得 $x_n\to x_0(n\to\infty)$. 现证 $\{f_1(x_n)\}$ 收敛. 事实上，由于 $f_1(x)$ 在 $E_1$ 上一致连续：$\forall\varepsilon>0,\exists\delta>0$，当 $x',x''\in E_1,|x'-x''|<\delta$ 时，有 $|f_1(x')-f_1(x'')|<\varepsilon$. 由 $\{x_n\}$ 的收敛性，对此 $\delta>0,\exists N>0$，当 $m,n>N$ 时，有 $|x_m-x_n|<\delta$，从而 $|f_1(x_m)-f_1(x_n)|<\varepsilon$. 故 $\{f_1(x_n)\}$ 收敛.

$2^\circ$ 另设 $x_n'\in E_1(n=1,2,\cdots),x_n'\to x_0(n\to\infty)$，需证明 $\lim\limits_{n\to\infty}f_1(x_n')=\lim\limits_{n\to\infty}f_1(x_n)$. 但这是明显的，否则令 $\bar x_{2n}=x_n$ 及 $\bar x_{2n-1}=x_n'(n=1,2,\cdots)$，有 $\bar x_n\in E_1,\bar x_n\to x_0$ 使得 $\{f_1(\bar x_n)\}$ 发散，这与 $1^\circ$ 的结论矛盾.

至此，我们可以在 $E$ 上任意地定义函数
\[
  f(x_0)=\lim_{n\to\infty}f_1(x_n)\quad(\forall x_0\in E), \tag{1}
\]
其中 $\{x_n\}$ 是 $E_1$ 中任意一个趋于 $x_0$ 的序列.

$3^\circ$ 我们来证明 $f(x)$ 满足条件 1）、2）. 事实上，$\forall x\in E_1$，只要令 $x_n=x(n=1,2,\cdots)$，则
\[
  f(x)=\lim_{n\to\infty}f_1(x_n)=f_1(x).
\]
这就证明了条件 1）.

为了证明 $f(x)$ 适合条件 2），我们只要证明：“$f(x)$ 在 $E$ 的任何有界子集 $F\subset E$ 上一致连续.” 这是因为，对于任意 $x_0\in E$，我们总可找到包含 $x_0$ 的有界子集 $F\subset E$，既然 $f$ 在 $F$ 上一致连续，自然 $f$ 在 $x_0$ 处连续. 然后由 $x_0$ 的任意性，知 $f$ 在 $E$ 上处处连续. 其次，若 $E$ 本身为有界集合，则可令 $F=E$，便知 $f$ 在 $E$ 上一致连续. 剩下的问题在于证明 $f(x)$ 在 $F$ 上一致连续. 即要证明：$\forall\varepsilon>0,\exists\delta>0$，当 $x',x''\in F,|x'-x''|<\delta$ 时，有 $|f(x')-f(x'')|<\varepsilon$. 实际上，已知 $f_1(x)$ 在 $E_1$ 上一致连续，所以 $\forall\varepsilon>0,\exists\delta_1>0$，当 $x',x''\in E_1,|x'-x''|<\delta_1$ 时，有
\[
  |f_1(x')-f_1(x'')|<\frac\varepsilon3. \tag{2}
\]
令取 $\delta=\dfrac{\delta_1}{3}$，可以证明这时若 $x',x''\in F,|x'-x''|<\delta$，则有 $|f(x')-f(x'')|<\varepsilon$. 因为对 $x',x''\in F$，$\exists x_n',x_n''\in E_1(n=1,2,\cdots)$，使得 $x_n'\to x',x_n''\to x''$（当 $n\to\infty$ 时），$f(x')=\lim\limits_{n\to\infty}f_1(x_n'),f(x'')=\lim\limits_{n\to\infty}f_1(x_n'')$.

从而对于 $\varepsilon>0$ 与 $\delta_1>0,\exists n_0$（充分大）使得 $|x_{n_0}'-x'|<\dfrac{\delta_1}{3}$，$|x_{n_0}''-x''|<\dfrac{\delta_1}{3}$，
\[
  |f(x')-f_1(x_{n_0}')|<\frac\varepsilon3,\quad
  |f(x'')-f_1(x_{n_0}'')|<\frac\varepsilon3.
\]
故
\[
  |x_{n_0}'-x_{n_0}''|\le |x_{n_0}'-x'|+|x'-x''|+|x''-x_{n_0}''|
  <\frac{\delta_1}{3}+\frac{\delta_1}{3}+\frac{\delta_1}{3}=\delta_1.
\]
于是由式 (2)，有 $|f_1(x_{n_0}')-f_1(x_{n_0}'')|<\dfrac\varepsilon3$. 因此
\[
\begin{aligned}
|f(x')-f(x'')|
&\le |f(x')-f_1(x_{n_0}')|+|f_1(x_{n_0}')-f_1(x_{n_0}'')|+|f_1(x_{n_0}'')-f(x'')|\\
&<\frac\varepsilon3+\frac\varepsilon3+\frac\varepsilon3=\varepsilon.
\end{aligned}
\]
问题证毕.

\noindent\textbf{$\star$2.2.1}\quad 设 $f$ 是区间 $I$ 上的实函数，试证如下三条件有逻辑关系：1）$\Rightarrow$2）$\Rightarrow$3）.

1）$f$ 在 $I$ 上可导且导函数有界，即：$\exists M>0$ 使得 $|f'(x)|\le M(\forall x\in I)$；

2）$f$ 在 $I$ 上满足 Lipschitz 条件，即：$\exists L>0$ 使得 $|f(x')-f(x'')|\le L|x'-x''|(\forall x',x''\in I)$；

3）$f$ 在 $I$ 上一致连续.

\noindent\textbf{提示}\quad （1）$\Rightarrow$2））用 Lagrange 中值公式，取 $L=M$；（2）$\Rightarrow$3））取 $\delta=\dfrac\varepsilon L$.

\noindent\textbf{$\star$2.2.2}\quad 设 $f(x)$ 在区间 $I$ 上有定义. 为了检验 $f$ 在 $I$ 上是否一致连续，今设计如下的实验：取一根内空直径为 $\varepsilon$ 的圆形直管（$\varepsilon>0$），截取长度为 $\delta$ 的一段（$\delta>0$），将直管中轴与 $x$ 轴平行放好，然后让 $y=f(x)$ 的曲线平移从管内穿过. 若不论 $\varepsilon>0$ 多么小，只要事先将直管长度 $\delta>0$ 取定足够短，曲线就能平移穿过此管，整个穿越过程 $\delta$ 无须改变，那么 $f$ 就在 $I$ 上一致连续；否则就是非一致连续. 问这种理解正确吗？（注 一致性主要体现在“整个穿越过程 $\delta$ 无须改变”上！）\hfill《正确》

\noindent\textbf{$\star$2.2.3}\quad 函数 $f(x)$ 在 $[a,b]$ 上一致连续，又在 $[b,c]$ 上一致连续，$a<b<c$. 用定义证明：$f(x)$ 在 $[a,c]$ 上一致连续.（北京大学）

\section*{※ §2.3\quad 上、下半连续}

为了扩大眼界，本节对半连续函数作简短介绍，供有兴趣的读者阅读. 想了解详情可查看本书第 1 或 2 版. 运用前面已学过的知识和方法，从半连续的定义出发，建立跟连续函数平行的理论，没有原则性的困难. 下面列举的结果，读者可作为练习，独立完成相关证明.

\subsection*{一、上、下半连续的定义及等价条件}

\noindent\textbf{定义}\quad 设函数 $f(x)$ 在集合 $E$ 上有定义，$x_0\in E$ 为 $E$ 的一个聚点.

1）若 $\forall\varepsilon>0,\exists\delta>0$，当 $x\in E,|x-x_0|<\delta$ 时，恒有 $f(x)<f(x_0)+\varepsilon$，则称 $f(x)$ 在 $x_0$ 处上半连续.

2）若 $\forall\varepsilon>0,\exists\delta>0$，当 $x\in E,|x-x_0|<\delta$ 时，恒有 $f(x)>f(x_0)-\varepsilon$，则称 $f(x)$ 在 $x_0$ 处下半连续.

（明显）若 $f(x)$ 在点 $x_0$ 连续，则 $f(x)$ 在点 $x_0$ 既上半连续，又下半连续.

\noindent\textbf{例}\quad Dirichlet 函数
\[
  D(x)=
  \begin{cases}
    1, & \text{当 }x\text{ 为有理数时},\\
    0, & \text{当 }x\text{ 为无理数时}
  \end{cases}
\]
在有理点处上半连续，但不下半连续；在无理点的情况恰恰相反.

函数 $f(x)=xD(x)$，当 $x>0$ 时，跟 $D(x)$ 的结论一样；当 $x<0$ 时，跟 $D(x)$ 的结论相反；在 $x=0$ 处，$f(x)$ 连续（既上半连续，又下半连续）.

Riemann 函数 $R(x)$（见例 3.1.1）在无理点处连续（既上半连续又下半连续），但在有理点处只上半连续不下半连续.

\noindent\textbf{定理 1（等价描述之一）}\quad 设函数 $f(x)$ 在集合 $E$ 上有定义，$x_0$ 为 $E$ 的一个聚点，则如下断言等价：

i）$f(x)$ 在 $x_0$ 处上半连续；

ii）$\overline{\lim\limits_{x\to x_0}}f(x)\le f(x_0)$；

iii）$\forall\{x_n\}\subset E$，若 $x_n\to x_0$，则 $\overline{\lim\limits_{n\to\infty}}f(x_n)\le f(x_0)$.

\noindent\textbf{定理 2（等价描述之二）}\quad 设 $E$ 为闭集，$f(x)$ 在 $E$ 上有定义，则
\[
  f(x)\text{ 在 }E\text{ 中上半连续}
  \Longleftrightarrow \forall c\in(-\infty,+\infty),F(c)=\{x\in E\mid f(x)\ge c\}\text{ 为闭集}.
\]
该定理无须 $\varepsilon-\delta$ 语言，这为推广到点集拓扑作了准备.

\subsection*{二、上、下半连续的性质}

上、下半连续具有对偶性，下面只写出上半连续的结果，下半连续的相应结果读者可以自己写出.

\noindent\textbf{定理 3（运算性质）}\quad （在区间上半连续，意指：函数在区间上每点都上半连续.）

1）若在 $[a,b]$ 上函数 $f(x),g(x)$ 上半连续，则 $f(x)+g(x)$ 亦在 $[a,b]$ 上上半连续；

2）若在 $[a,b]$ 上函数 $f(x)$ 上半连续，则 $-f(x)$ 下半连续；

3）若在 $[a,b]$ 上函数 $f(x),g(x)$ 同为正值函数且上半连续（或同为负值函数且下半连续），则它们的积 $f(x)\cdot g(x)$ 亦上半连续；

4）若在 $[a,b]$ 上 $f(x)>0$ 且 $f(x)$ 上半连续，则 $\dfrac1{f(x)}$ 下半连续.

以上结论可根据半连续的定义（或等价述）直接推出.

\noindent\textbf{注}\quad （1）连续性有保号性，但上半连续只保负（未必保正），即：若 $f(x)$ 在 $x_0$ 处上半连续，且 $f(x_0)<0$，则在 $x_0$ 的充分小的邻域内恒有 $f(x)<0$.

类似地，下半连续只保正（未必保负）.

（2）半连续无介值性. 例如，
\[
  f(x)=
  \begin{cases}
    0, & x\in[0,1),\\
    2, & x\in[1,2]
  \end{cases}
\]
在 $[0,2]$ 上上半连续，但显然没有介值性（事实上 $(0,2)$ 内的值都未被取到）.

（3）（确界定理）若 $f(x)$ 上半连续，则 $f(x)$ 必有上界（未必有下界），且必定达到上确界.

\noindent\textbf{定理 4}\quad 若在 $(a,b)$ 内函数 $f(x)$ 上（或下）半连续，则存在内闭区间 $[\alpha,\beta]\subset(a,b)$，使得在 $[\alpha,\beta]$ 上 $f(x)$ 有界.

\noindent\textbf{定理 5（保半连续性）}\quad 设在集合 $E$ 上，$\{f_n(x)\}$ 是上半连续函数序列 $(n=1,2,\cdots)$，$|f_n(x)|$ 关于 $n$ 递减，即
\[
  f_1(x)\ge f_2(x)\ge\cdots\ge f_n(x)\ge f_{n+1}(x)\ge\cdots\quad(\forall x\in E),
\]
那么，若 $\lim\limits_{n\to\infty}f_n(x)=f(x)(\forall x\in E)$，则在 $E$ 上 $f(x)$ 也上半连续.

同样，收敛的下半连续函数之上升序列，其极限函数若存在，也必下半连续.

\noindent\textbf{定理 6（定理 5 的反问题）}\quad 在 $[a,b]$ 上，若 $f(x)$ 是上半连续函数，则必存在一个递减的连续函数序列 $\{f_n(x)\}$，使得 $\lim\limits_{n\to\infty}f_n(x)=f(x)$.

对下半连续有类似结论（希望感兴趣的读者，写出更多更好的结果）.

\section*{※ §2.4\quad 函数方程}

\noindent\textbf{导读}\quad 本节虽然不是重点，但例题中的方法十分精彩，值得学习和借鉴，适合各类读者，习题可作机动.

\subsection*{一、问题的提出}

若 $f(x)=ax$，则
\[
  f(x+y)=a(x+y)=ax+ay=f(x)+f(y)\quad(\forall x,y\in\mathbb{R}).
\]
这就是说，一次齐次函数 $f(x)=ax$ 满足函数方程
\[
  f(x+y)=f(x)+f(y)\quad(\forall x,y\in\mathbb{R}). \tag{A}
\]
同样，容易验证函数 $f(x)=a^x(a>0),g(x)=\log_a x(a>0),h(x)=x^a,i(x)=\cos ax$ 与 $j(x)=\operatorname{ch} ax$ 分别满足方程：
\begin{align}
f(x+y)&=f(x)\cdot f(y)\quad(\forall x,y\in\mathbb{R}), \tag{B}\\
f(x\cdot y)&=f(x)+f(y)\quad(\forall x,y>0), \tag{C}\\
f(x\cdot y)&=f(x)\cdot f(y)\quad(\forall x,y>0), \tag{D}\\
f(x+y)+f(x-y)&=2f(x)f(y)\quad(\forall x,y\in\mathbb{R}). \tag{E}
\end{align}
而函数 $f(x)=\sin ax$ 与 $g(x)=\cos ax$ 满足联立方程组
\[
\begin{cases}
f(x+y)=f(x)f(y)-g(x)g(y),\\
g(x+y)=f(x)g(y)+f(y)g(x)
\end{cases}
\quad(\forall x,y\in\mathbb{R}). \tag{F}
\]
现在我们要提出相反的问题：满足这些方程，是否仅是上述这些函数？下面将看到，若对求解的范围作连续等条件的限制，回答是肯定的；否则不一定.

值得注意的是，此问题的解法颇有启发性，耐人寻味. 上述五个方程，首先为 Cauchy 所研究，并给出了连续解.

\subsection*{二、求解函数方程}

\noindent\textbf{a. 推归法}

\noindent\textbf{要点}\quad 为了求解函数方程，我们可以从函数方程出发，由最简单的情况入手，一步一步地推导，边推导边归纳，逐步达到所希望的结果.

\noindent\textbf{例 2.4.1}\quad 函数方程 (A)：
\[
  f(x+y)=f(x)+f(y)\quad(\forall x,y\in\mathbb{R})
\]
在 $x=0$ 处连续的唯一解为 $f(x)=ax$（其中 $a$ 为常数）.

\noindent\textbf{证}\quad $1^\circ$（首先证明：$\forall c\in(-\infty,+\infty),f(cx)=cf(x)$.）

\[
  f(2x)=f(x+x)\overset{\text{式(A)}}{=}f(x)+f(x)=2f(x),\cdots;
\]
若 $f[(n-1)x]=(n-1)f(x)$，则
\[
\begin{aligned}
f(nx)
&=f[(n-1)x+x]\overset{\text{式(A)}}{=}f[(n-1)x]+f(x)\\
&=(n-1)f(x)+f(x)=nf(x).
\end{aligned}
\]
至此证明了：
\[
  f(nx)=nf(x)\quad(\forall n\in\mathbb{N}). \tag{1}
\]
在此式中用 $\dfrac xn$ 代换 $x$，则得 $f(x)=nf\left(\dfrac xn\right)$，
\[
  f\left(\frac1nx\right)=\frac1nf(x)\quad(\forall n\in\mathbb{N}). \tag{2}
\]
应用式 (1) 和 (2) 可得
\[
  f\left(\frac mnx\right)=\frac mn f(x)\quad(\forall n,m\in\mathbb{N}). \tag{3}
\]
又因为 $f(x)=f(0+x)=f(0)+f(x)$，故 $f(0)=0$. 从而
\[
  f(x)+f(-x)=f(x-x)=f(0)=0,
\]
这就得到了
\[
  f(-x)=-f(x). \tag{4}
\]
用 $\dfrac mnx$ 取代式 (4) 中的 $x$，得
\[
  f\left(-\frac mnx\right)=-\frac mn f(x)\quad(\forall n,m\in\mathbb{N}). \tag{5}
\]
以上我们利用式 (A) 证明了 $c$ 为有理数时，$f(cx)=cf(x)$. 但任何无理数总可以表示成有理数序列的极限，所以 $\forall c$（无理数），$\exists\{c_n\}$（有理数序列），使得 $c_n\to c$（当 $n\to\infty$ 时）. 于是
\[
\begin{aligned}
f(cx)-c_nf(x)
&=f(cx)-f(c_nx)\overset{\text{式(A)}}{=}f(cx-c_nx)\\
&=f[(c-c_n)x].
\end{aligned}
\]
令 $n\to\infty$，取极限，因 $f$ 在 $x=0$ 处连续，$f[(c-c_n)x]\to f(0)=0(n\to\infty)$，得
\[
  f(cx)=cf(x)\quad(\forall c\in\mathbb{R}). \tag{6}
\]

$2^\circ$ 由式 (6) 知，$\forall x\in\mathbb{R},f(x)=f(x\cdot1)=xf(1)$. 记 $a=f(1)$，则 $f(x)=ax(\forall x\in\mathbb{R})$. 证毕.

\noindent\textbf{例 2.4.2}\quad 函数方程 (E)：$f(x+y)+f(x-y)=2f(x)f(y)(\forall x,y\in\mathbb{R})$ 在实轴 $\mathbb{R}$ 上不恒为零的连续解为 $f(x)=\cos ax$ 或 $f(x)=\operatorname{ch} ax$（$a$ 为常数）

\noindent\textbf{证}\quad $1^\circ$（先证 $f(0)=1$，且 $f(x)$ 为偶函数.）在函数方程 (E) 中令 $y=0$，得 $2f(x)=2f(x)f(0)$. 因 $f(x)\ne0$，所以
\[
  f(0)=1; \tag{1}
\]
在 (E) 中令 $x=0$，得 $f(y)+f(-y)=2f(y)$. 所以，$\forall x\in\mathbb{R}$，有
\[
  f(-x)=f(x). \tag{2}
\]
表明 $f(x)$ 为偶函数.

$2^\circ$（用推归法证明解为 $\cos ax$ 或 $\operatorname{ch} ax$.）因 $f(0)=1$，根据连续函数局部保号性，$\exists c>0$，使得 $x\in[0,c]$ 时，有 $f(x)>0$. 下面分两种情况讨论：

（a）若 $f(c)\le1$. 由 $0<f(c)\le1$，知 $\exists\theta\in\left[0,\dfrac\pi2\right]$，使得
\[
  f(c)=\cos\theta. \tag{3}
\]
（下面我们用推归法证明 $\forall x\in\mathbb{R}$，有 $f(x)=\cos\theta x$.）

将方程 (E) 写成
\[
  f(x+y)=2f(x)f(y)-f(x-y). \tag{4}
\]
令 $x=y=c$，得
\[
  f(2c)=2(f(c))^2-f(0)=2\cos^2\theta-1=\cos2\theta.
\]
如此我们已得
\[
  f(c)=\cos\theta,\quad f(2c)=\cos2\theta.
\]
若已得
\[
  f[(n-2)c]=\cos(n-2)\theta,\quad f[(n-1)c]=\cos(n-1)\theta.
\]
在式 (4) 中令 $x=(n-1)c,y=c$，则得
\[
  f(nc)=2\cos(n-1)\theta\cdot\cos\theta-\cos(n-2)\theta=\cos n\theta.
\]
故 $\forall n\in\mathbb{N}$，有
  \[
    f(nc)=\cos n\theta. \tag{5}
  \]
将 (E) 写成
\[
  f(x)f(y)=\frac12[f(x+y)+f(x-y)]. \tag{6}
\]
令 $x=y=\dfrac c2$，代入得
\[
  \left[f\left(\frac c2\right)\right]^2=\frac12(\cos\theta+1)=\left(\cos\frac12\theta\right)^2.
\]
注意到 $\dfrac c2\in[0,c]$，所以 $f\left(\dfrac c2\right)>0$，故 $f\left(\dfrac c2\right)=\cos\dfrac12\theta$.

用数学归纳法：若已有 $f\left(\dfrac c{2^{n-1}}\right)=\cos\dfrac{\theta}{2^{n-1}}$，则在式 (6) 中令 $x=y=\dfrac c{2^n}$，便得
\[
  \left[f\left(\frac c{2^n}\right)\right]^2
  =\frac12\left[f\left(\frac c{2^{n-1}}\right)+f(0)\right]
  =\frac12\left(\cos\frac{\theta}{2^{n-1}}+1\right)
  =\left(\cos\frac{\theta}{2^n}\right)^2.
\]
因 $f\left(\dfrac c{2^n}\right)$ 为正，所以
\[
  f\left(\frac c{2^n}\right)=\cos\frac{\theta}{2^n}. \tag{7}
\]
这就证明了式 (7) 对一切 $n\in\mathbb{N}$ 成立.

利用式 (5) 和 (7)，有
\[
  f\left(\frac m{2^n}c\right)=\cos\frac m{2^n}\theta,\quad \forall m,n\in\mathbb{N}. \tag{8}
\]
但任何 $x>0$，总可写成 $\dfrac m{2^n}$ 形式的实数的极限\footnote{因 $\left\{\dfrac m{2^n}\mid m,n\in\mathbb{N}\right\}$ 在半轴 $x>0$ 上处处稠密.}，即 $\forall x>0,\exists x_i(i=1,2,\cdots)$（$x_i$ 为形如 $\dfrac m{2^n}$ 的数），使得 $x_i\to x(i\to\infty)$. 则 $f(x_ic)=\cos x_i\theta$. 令 $i\to+\infty$，由连续性得
\[
  f(cx)=\cos\theta x.
\]
再注意到式 (1) 与 (2)，可知此式对于 $x\le0$ 亦成立. 最后，将此式中的 $x$ 换为 $\dfrac xc$，并记 $\dfrac\theta c=a$，则得
\[
  f(x)=\cos ax\quad(\forall x\in\mathbb{R}).
\]
（b）若 $f(c)>1$. 类似可证 $f(x)=\operatorname{ch} ax$（留作练习）.

以上我们看到，推归法是一种构造性的方法，对某些函数方程，应用此法十分有效. 但这些毕竟是麻烦的. 能有更简便的方法，尽量用.

\noindent\textbf{b. 转化法}

\noindent\textbf{要点}\quad 引入适当的新自变量或新因变量，作变量替换，将所给的函数方程转化为熟知的函数方程；或从所给的限制条件，导出熟知的限制条件；从而利用熟知的问题求解.

\noindent\textbf{例 2.4.3}\quad 证明：在实轴 $\mathbb{R}$ 上满足方程 (B)
\[
  f(x+y)=f(x)f(y)
\]
的唯一不恒等于零的连续函数是 $f(x)=a^x(a>0\text{ 为常数})$.

\noindent\textbf{证}\quad $1^\circ$（证明 $f(x)>0$.）因 $f(x)\ne0$，故 $\exists x_0\in\mathbb{R}$，使得 $f(x_0)\ne0$. 于是，$\forall x\in\mathbb{R}$，由于
\[
  f(x)f(x_0-x)\overset{\text{式(B)}}{=}f[x+(x_0-x)]=f(x_0)\ne0,
\]
知 $f(x)\ne0$. 从而
\[
  f(x)=f\left(\frac x2+\frac x2\right)\overset{\text{式(B)}}{=}\left[f\left(\frac x2\right)\right]^2>0.
\]

$2^\circ$（作变换）令 $F(x)=\log_a f(x)$，由 $1^\circ$ 知 $f(1)>0$. 不妨取 $a=f(1)$. 于是 $F(x)$ 连续，且满足方程 (A)：
\[
  F(x+y)=F(x)+F(y)\quad(\forall x,y\in\mathbb{R}).
\]
故由例 2.4.1，$F(x)=a_1x$（其中 $a_1$ 为常数），但 $F(1)=\log_a f(1)=\log_a a=1$，所以 $a_1=F(1)=1$，故 $F(x)=x$. 从而
\[
  f(x)=a^{F(x)}=a^x(a=f(1)>0\text{ 为常数}).
\]

\noindent\textbf{例 2.4.4}\quad 证明：满足方程 (C)：
\[
  f(x\cdot y)=f(x)+f(y)\quad(\forall x,y>0)
\]
唯一不恒等于零的函数为 $f(x)=b\log_a x(a>0,a\text{ 和 }b\text{ 为常数})$.

\noindent\textbf{提示}\quad 令 $g(x)=f(a^x)$，利用例 2.4.1.

\noindent\textbf{例 2.4.5}\quad 求在 $\mathbb{R}$ 上满足方程
\[
  f\left(\frac{x+y}{2}\right)=\frac{f(x)+f(y)}2\quad(\forall x,y\in\mathbb{R}) \tag{1}
\]
的连续函数.

\noindent\textbf{解}\quad 利用方程 (1)，
\[
  \frac{f(x)+f(y)}2
  =f\left(\frac{x+y}{2}\right)
  =f\left(\frac{\frac{x+y}{2}+0}{2}\right)
  =\frac{f(x+y)+f(0)}2
  \overset{f(0)=b}{=}\frac{f(x+y)+b}2.
\]
因此，$f(x)+f(y)=f(x+y)+b$，即
\[
  f(x)-b+f(y)-b=f(x+y)-b.
\]
由此令 $g(x)=f(x)-b,g(x)$ 满足例 2.4.1 中的方程 (A)：
\[
  g(x+y)=g(x)+g(y),
\]
所以 $g(x)=ax$，从而
\[
  f(x)=g(x)+b=ax+b\quad(\text{其中 }a,b\text{ 为常数}).
\]
不难验算 $f(x)$ 是方程 (1) 的解.

\noindent\textbf{例 2.4.6}\quad 设 $\Delta f(x)=f(x+\Delta x)-f(x)$，
\[
  \Delta^2f(x)=\Delta(\Delta f(x))=f(x+2\Delta x)-2f(x+\Delta x)+f(x).
\]
试求满足方程 $\Delta^2f(x)=0(\forall x\in\mathbb{R})$ 的连续函数.

\noindent\textbf{解}\quad 因 $f(x+2\Delta x)-2f(x+\Delta x)+f(x)=0$，故
\[
  f(x+\Delta x)=\frac{f(x+2\Delta x)+f(x)}2.
\]
令 $x+2\Delta x=y$，则得 $f\left(\dfrac{x+y}{2}\right)=\dfrac{f(x)+f(y)}2$. 由上例可知 $f(x)=ax+b$（其中 $a,b$ 为常数）. 不难验算此函数满足所给的方程.

上面这些例题都是通过变换转化方程. 下面再看另一种转化.

\noindent\textbf{例 2.4.7}\quad 函数方程 (A)：$f(x+y)=f(x)+f(y)(\forall x,y\in\mathbb{R})$ 在区间 $(-\eta,\eta)$ 内有界的唯一解为 $f(x)=ax$（其中 $\eta>0$ 为某常数，$a=f(1)$）.

\noindent\textbf{分析}\quad 本例的方程与例 2.4.1 的方程一样，只条件不同：那里要求在 $x=0$ 处连续，这里要求函数在 $(-\eta,\eta)$ 内有界. 事实上，由后者可以导出前者. 因为在例 2.4.1 中，我们看到由方程 (A) 可以推出：$\forall n\in\mathbb{N}$，有 $f\left(\dfrac1nx\right)=\dfrac1nf(x)$ 及 $f(0)=0$. 于是，由方程 (A) 可得
\[
  |f(x)-f(0)|=|f(x)|=\left|f\left(\frac1nnx\right)\right|=\frac1n|f(nx)|. \tag{1}
\]
因 $f(x)$ 在 $(-\eta,\eta)$ 内有界，即：$\exists M>0$，当 $-\eta<x<\eta$ 时有 $|f(x)|\le M$. $\forall\varepsilon>0$，令 $n>\dfrac M\varepsilon$，取 $\delta=\dfrac\eta n$，则 $|x|<\delta$ 时 $|nx|<\eta$. 由式 (1) 知
\[
  |f(x)-f(0)|\le\frac Mn<\varepsilon,
\]
故 $f$ 在 $x=0$ 处连续. 利用例 2.4.1 结果，其解自得.

\noindent\textbf{c. 利用微分方程}

\noindent\textbf{例 2.4.8}\quad 设函数 $f(x)$ 连续，$f'(0)$ 存在，并且对于任何 $x,y\in\mathbb{R}$，
\[
  f(x+y)=\frac{f(x)+f(y)}{1-4f(x)f(y)}. \tag{1}
\]

1）证明：$f(x)$ 在 $\mathbb{R}$ 上可微；

2）若 $f'(0)=\dfrac12$，求 $f(x)$.（中国人民大学）

\noindent\textbf{分析}\quad 式 (1) 中令 $x=y=0$，立即看出 $f(0)=0$. 要证 $f(x)$ 在 $\mathbb{R}$ 上可微，即 $\forall x\in\mathbb{R}$，要证如下极限存在：
\[
  f'(x)=\lim_{y\to0}\frac{f(x+y)-f(x)}{y}
  \overset{\text{式(1)}}{=}\lim_{y\to0}\frac{\frac{f(y)-0}{y}\cdot[1+4f^2(x)]}{1-4f(x)f(y)}.
\]
但已知 $f'(0)$ 存在，$f$ 连续，且已得 $f(0)=0$，故 $f'(x)=f'(0)[1+4f^2(x)]$. 因此

1）$f$ 在 $\mathbb{R}$ 上处处可微.

2）$f'(0)=\dfrac12$ 时，解微分方程
\[
  \begin{cases}
    y'=\dfrac12(1+4y^2),\\
    y|_{x=0}=0,
  \end{cases}
\]
可得 $y=\dfrac12\tan x$.

\section*{单元练习 2.4}

\noindent\textbf{2.4.1}\quad 1）设函数 $f(x)$ 在 $(0,+\infty)$ 上满足
\[
  f(2x)=f(x)\quad\text{且}\quad \lim_{x\to+\infty}f(x)=A.
\]
证明：$f(x)\equiv A,x\in(0,+\infty)$；（天津大学，湖北大学）

\noindent\textbf{new 2）}\quad 设函数 $f(x)$ 在 $(-\infty,+\infty)$ 上满足
\[
  f(2x)=f(x)\cos x\quad\text{及}\quad \lim_{x\to0}f(x)=f(0)=1,
\]
求 $f(x)$.（华中师范大学）\hfill$\left(f(x)=\dfrac{\sin x}{x}\text{（当 }x\ne0\text{ 时）},f(0)=1\right)$

\noindent\textbf{提示}\quad 1）$\forall x_0>0$，有 $f(x_0)=f(2x_0)=f(2^2x_0)=\cdots$.

2）方法 I\quad 反复迭代得 $f(x)=f\left(\dfrac{x}{2^n}\right)\cos\dfrac{x}{2^n}\cos\dfrac{x}{2^{n-1}}\cdots\cos\dfrac x2$，利用例 1.3.2 中 1）的极限：
\[
  \lim_{n\to\infty}\cos\frac{x}{2^n}\cos\frac{x}{2^{n-1}}\cdots\cos\frac x2=\frac{\sin x}{x}.
\]

\noindent\textbf{方法 II}\quad $f(2x)=f(x)\cos x\Rightarrow f(x)=f\left(\dfrac x2\right)\cos\dfrac x2=f\left(\dfrac x2\right)\dfrac{\sin x}{2\sin\frac x2}$，反复迭代，得 $f(x)=f\left(\dfrac{x}{2^n}\right)\dfrac{\sin x}{2^n\sin\frac{x}{2^n}}\to\dfrac{\sin x}{x}$（当 $n\to\infty$ 时）.

\noindent\textbf{2.4.2}\quad 试用推归法重新证明例 2.4.3 与例 2.4.4.

\noindent\textbf{2.4.3}\quad 证明：在 $\mathbb{R}$ 上满足方程
\[
  f(x+y)=f(x)+f(y)\quad(\forall x\in\mathbb{R})
\]
的唯一单调函数是 $f(x)=ax$（其中 $a$ 为常数）.

\noindent\textbf{提示}\quad 参考例 2.4.1，注意：“连续”条件，这里换为了“单调”条件.

\noindent\textbf{再提示}\quad 照样可推出 $f(cx)=cf(x)$，当 $c$ 为有理数时已证明成立. 要证明对无理数也成立，可按 $f$ 递增（或递减）两种情况分别讨论：例如，当 $f$ 递增时，设 $x>0$（$x<0$ 时类似可证）（$x=0$ 时知 $f(0)=0$，因此 $f(cx)=cf(x)$ 当 $x=0$ 时明显成立），取递增的有理数序列 $\{c_n\}:c_n<c,c_n\nearrow c$，则 $\{f(c_nx)\}\nearrow$，且有上界 $f(cx)$. 再在不等式 $f(c_nx)<f(cx)$ 里取极限，得
\[
  \lim_{n\to\infty}f(c_nx)=\lim_{n\to\infty}c_nf(x)=cf(x)\le f(cx). \tag{1}
\]
同样，可取有理数列 $\{c_n'\}:c_n'>c,c_n'\searrow c$，可得
\[
  cf(x)\ge f(cx). \tag{2}
\]
故
\[
  cf(x)=f(cx)\quad(\forall c,x\in\mathbb{R}).
\]
$f$ 递减，类似可证. 于是
\[
  f(x)=f(x\cdot1)=xf(1)\overset{\text{记 }a=f(1)}{=}ax.
\]

\noindent\textbf{2.4.4}\quad 证明：若 $f(x)$ 在 $\mathbb{R}$ 上满足方程 $f(x+y)=f(x)+f(y)$，则如下三条件等价：

1）$f(x)$ 在 $x=0$ 处连续；\qquad
2）$f(x)$ 在 $\mathbb{R}$ 上连续；\qquad
3）$\exists\delta>0,f(x)$ 在 $(-\delta,\delta)$ 上有界.

\noindent\textbf{提示}\quad 证明 1）$\Rightarrow$2）$\Rightarrow$3）$\Rightarrow$1）即可（可参考例 2.4.7）.

\noindent\textbf{再提示}\quad
\[
  f(x+h)=f(x)+f(h)\overset{x=h=0}{\Longrightarrow}f(0)=0. \tag{1}
\]
（1）$\Rightarrow$2））$f$ 在 $x=0$ 处连续 $\Rightarrow\lim\limits_{h\to0}f(h)=f(0)\overset{\text{式(1)}}{=}0$
\[
\Rightarrow \forall x\in\mathbb{R},\lim_{h\to0}f(x+h)=\lim_{h\to0}[f(x)+f(h)]=f(x)+\lim_{h\to0}f(h)=f(x),
\]
即 $f(x)$ 在 $\mathbb{R}$ 上连续，条件 2）成立.

（2）$\Rightarrow$3））$\mathbb{R}$ 上连续 $\Rightarrow[-\delta,\delta]$ 上连续 $\Rightarrow[-\delta,\delta]$ 上有界 $\Rightarrow(-\delta,\delta)$ 上有界.

（3）$\Rightarrow$1））（反证法）若 1）不成立，则 $\exists\varepsilon_0>0$，使得
\[
  \forall n\in\mathbb{N},\exists x_n:\ |x_n|<\frac1n,\text{ 但 }|f(x_n)|\ge\varepsilon_0. \tag{2}
\]
因此
\[
\begin{aligned}
|f(nx_n)|
&=|f[x_n+(n-1)x_n]|=|f(x_n)+f[(n-1)x_n]|\\
&=|2f(x_n)+f[(n-2)x_n]|=\cdots=n|f(x_n)|\overset{\text{式(2)}}{>}n\varepsilon_0\to+\infty\quad(n\to\infty).
\end{aligned}
\]
与已知条件“$f$ 在 $(-\delta,\delta)$ 有界”矛盾.

\noindent\textbf{注}\quad 3）$\Rightarrow$1）也可不用反证法，见例 2.4.7 里的“分析”.

\noindent\textbf{2.4.5}\quad 证明：若 $f(x)$ 在 $\mathbb{R}$ 上连续，对任意 $x,y\in\mathbb{R}$，有 $f(x+y)=f(x)\cdot f(y)$，则 $f(x)$ 在 $\mathbb{R}$ 上可微.（东北师范大学）

\noindent\textbf{2.4.6}\quad 证明：满足方程 $f(xy)=f(x)f(y)(\forall x,y>0)$ 的唯一不恒等于 0 的连续函数是 $f(x)=x^a$（$a$ 为常数）.

\noindent\textbf{提示}\quad 对所给的方程取对数，再用例 2.4.4 中的变换方法转化为例 2.4.1.

\noindent\textbf{证}\quad $\forall x>0$，有 $f(x)=f(\sqrt{x}\sqrt{x})\overset{\text{所给方程}}{=}f^2(\sqrt{x})>0$（恒正）.

因此，取 $b>0$，对所给方程取对数得
\[
  \log_b f(xy)=\log_b f(x)+\log_b f(y). \tag{1}
\]
利用例 2.4.4 的变换方法，记 $g(x)=f(b^x)$. 由式 (1) 得
\[
  g(x+y)=f(b^{x+y})=f(b^x\cdot b^y).
\]
于是
\[
\begin{aligned}
\log_b g(x+y)
&=\log_b f(b^{x+y})=\log_b f(b^x\cdot b^y)\\
&=\log_b f(b^x)+\log_b f(b^y)=\log_b g(x)+\log_b g(y).
\end{aligned} \tag{2}
\]
式 (2) 说明 $\log_b g(x)$ 满足例 2.4.1 的条件，因此在 $x=0$ 处连续唯一解是
\[
  \log_b g(x)=ax\quad(a\text{ 为常数}),
\]
亦即 $\log_b f(b^x)=ax$ 或 $f(b^x)=b^{ax}=(b^x)^a$，故 $f(x)=x^a(a\text{ 为常数})$.

\noindent\textbf{2.4.7}\quad 求在 $\mathbb{R}$ 上满足方程 $f(xy)=f(x)f(y)(\forall x,y\in\mathbb{R})$ 的一切连续函数，并证明不连续函数 $f(x)=\operatorname{sgn} x$ 在 $\mathbb{R}$ 上也处处满足方程.

\noindent\textbf{提示}\quad 利用已知等式，
\[
\begin{gathered}
f(x)=f(x\cdot1)=f(x)f(1)\Rightarrow f(1)=1,\\
\Rightarrow1=f(1)=f((-1)\cdot(-1))=f(-1)\cdot f(-1)\Rightarrow[f(-1)]^2=1\\
\Rightarrow f(-1)=\pm1\Rightarrow f(-x)=f(-1)f(x)=\pm f(x)\quad(\forall x\in\mathbb{R}),
\end{gathered}
\]
即 $f(x)$ 为偶函数或奇函数. 根据上题，$x>0,f(x)=x^a(a\text{ 为常数})$.

因此在 $\mathbb{R}$ 上，

当 $f(x)$ 为偶函数时：$f(x)=|x|^a$（$a$ 为常数）；

当 $f(x)$ 为奇函数时：$f(x)=(\operatorname{sgn} x)|x|^a$（$a$ 为常数）.

取 $a=0$，得
\[
  f(x)=\operatorname{sgn} x=
  \begin{cases}
    1, & x>0,\\
    0, & x=0,\\
    -1, & x<0.
  \end{cases}
\]
$f(x)$ 在原点处不连续. 但代入验算，明显还满足方程 $f(xy)=f(x)f(y)$.

\noindent\textbf{2.4.8}\quad 设函数 $f(x),g(x)$ 在 $\mathbb{R}$ 上连续有界，满足方程组 $(\forall x,y\in\mathbb{R})$
\[
\begin{cases}
f(x+y)=f(x)f(y)-g(x)g(y),\\
g(x+y)=f(x)g(y)+f(y)g(x)
\end{cases} \tag{1}
\]
及 $f(0)=1,g(0)=0$. 证明：$f(x)=\cos ax,g(x)=\pm\sin ax$（其中 $a$ 为常数）.

\noindent\textbf{证法 I（转化法）}\quad \textbf{提示}\quad 利用例 2.4.3 和例 2.4.2 的已有结果.

\noindent\textbf{再提示}\quad $1^\circ$ 令 $F(x)=f^2(x)+g^2(x)$，由式 (1)，(2) 可得 $F(x+y)=F(x)F(y)$，由例 2.4.3 可得 $F(x)=a^x$（其中 $a=F(1)>0$）. 但要求 $f(x),g(x)$ 有界，只能 $a=1$. 故
\[
  F(x)=f^2(x)+g^2(x)\equiv1\quad(\forall x\in\mathbb{R}). \tag{3}
\]

$2^\circ$
\[
\begin{aligned}
f(-x)
&=f(-x)\cdot1+g(-x)\cdot0\\
&=f(-x)\cdot f(0)+g(-x)\cdot g(0)\\
&=f(-x)\cdot(\text{式 }(1))_{y=-x}+g(-x)\cdot(\text{式 }(2))_{y=-x}\\
&=f(x)\quad(\forall x\in\mathbb{R}),
\end{aligned} \tag{4}
\]
即
\[
  f(-x)=f(x)\quad(\forall x\in\mathbb{R}).
\]
（表明 $f$ 是偶函数.）

同理，有
\[
  g(-x)=-g(x)\quad(\forall x\in\mathbb{R}). \tag{5}
\]
（表明 $g$ 是奇函数.）将式 (4), (5) 代入式 (1)，得函数 $f(x)$ 满足例 2.4.2 的条件：
\[
  f(x+y)+f(x-y)=2f(x)f(y)\quad(\forall x,y\in\mathbb{R}).
\]
故由例 2.4.2 得 $f(x)=\cos ax$；再由式 (3) 得 $g(x)=\pm\sin ax$（$a$ 为常数，$\forall x\in\mathbb{R}$）.

\noindent\textbf{证法 II（直接法）}\quad 沿用证法 I 中的式 (3)：$\forall x\in\mathbb{R},f^2(x)+g^2(x)=1$，知：$|f(x)|\le1,|g(x)|\le1$. 因此，对 $x=x_0\in\mathbb{R}$，$\exists\theta\in\left[-\dfrac\pi2,\dfrac\pi2\right]$，使得
\[
  f(x_0)=\cos\theta,\quad g(x_0)=\pm\sqrt{1-\cos^2\theta}=\pm\sin\theta.
\]
在式 (1) 和 (2) 里令 $x=y=x_0$，再利用上式，可得
\[
  f(2x_0)=f^2(x_0)-g^2(x_0)=\cos2\theta,\quad
  g(2x_0)=2f(x_0)g(x_0)=\pm\sin2\theta. \tag{6}
\]
（数学归纳法）设已证得
\[
  f(nx_0)=\cos n\theta,\quad g(nx_0)=\pm\sin n\theta\quad(\forall n\in\mathbb{N}),
\]
那么利用式 (1) 可得
\[
\begin{aligned}
f[(n+1)x_0]
&=f(nx_0+x_0)=f(nx_0)f(x_0)-g(nx_0)g(x_0)\\
&=\cos n\theta\cos\theta-\sin n\theta\sin\theta=\cos(n+1)\theta.
\end{aligned}
\]
同理，利用式 (2) 可得
\[
  g[(n+1)x_0]=\pm\sin(n+1)\theta.
\]
类似可证出：
\[
  f\left(\frac1n x_0\right)=\cos\left(\frac1n\theta\right),\quad
  g\left(\frac1n x_0\right)=\pm\sin\left(\frac1n\theta\right)\quad(\forall n\in\mathbb{N}).
\]
于是 $\forall c\in\mathbb{R}$，都有
\[
  f(cx_0)=\cos c\theta,\quad g(cx_0)=\pm\sin c\theta.
\]
将 $cx_0$ 记为 $x$，则 $c\theta=cx_0\cdot\dfrac{\theta}{x_0}=ax$（其中 $a=\dfrac{\theta}{x_0}$ 为常数）. 最后得
\[
  f(x)=\cos ax,\quad g(x)=\pm\sin ax\quad(a\text{ 为常数}).
\]

\]

\noindent\textbf{2.4.10}\quad 证明：满足方程
\[
  f(x+y)=\frac{f(x)+f(y)}{1-f(x)f(y)}\quad(\forall x,y\in\mathbb{R})
\]
的唯一可导函数是 $f(x)=\tan ax$（其中 $a$ 为常数）.

\noindent\textbf{提示}\quad 方法 I\quad 令 $f(x)=2g(x)$ 代入，则 $g$ 可利用例 2.4.8.

方法 II\quad 使用例 2.4.8 的方法，转化为微分方程.

\noindent\textbf{再提示}\quad （使用方法 II 时）在原式里令 $x=y=0$，可得 $f(0)=0$. 于是
\[
\begin{aligned}
\frac{f(x+h)-f(x)}{h}
&\overset{\text{已知方程}}{=}\frac1h\left(\frac{f(x)+f(h)}{1-f(x)f(h)}-f(x)\right)\\
&\overset{f(0)=0}{=}\frac{f(h)-f(0)}{h}\cdot\frac{1+f^2(x)}{1-f(x)f(h)}.
\end{aligned}
\]
再令 $h\to0$，取极限（记 $f'(0)=a$），则上式变为 $\dfrac{\mathrm{d}}{\mathrm{d}x}f(x)=a(1+f^2(x))$，亦即
\[
  \frac{\mathrm{d}f(x)}{1+f^2(x)}=a\,\mathrm{d}x.
\]
求积分得 $\arctan f(x)=ax+c$. 又因 $f(0)=0$，知 $c=0$，故 $f(x)=\tan ax$（$a$ 为常数）.

\noindent\textbf{new 2.4.12}\quad 设 $f$ 是实数轴上的可微函数，对任意 $x,y\in\mathbb{R}$，满足
